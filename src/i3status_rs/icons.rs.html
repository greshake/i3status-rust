10#[derive(Deserialize, Debug, Clone)]

12pub enum Icon {

15}

17impl From<&'static str> for Icon {
18    fn from(value: &'static str) -> Self {
19        Self::Single(value.into())
20    }
21}
22
23impl<const N: usize> From<[&str; N]> for Icon {
24    fn from(value: [&str; N]) -> Self {
25        Self::Progression(value.iter().map(|s| s.to_string()).collect())
26    }
27}

212impl Icons {
213    pub fn from_file(file: &str) -> Result<Self> {
214        if file == "none" {
215            Ok(Icons::default())
216        } else {
217            let file = util::find_file(file, Some("icons"), Some("toml"))?
218                .or_error(|| format!("Icon set '{file}' not found"))?;
219            Ok(Icons(util::deserialize_toml_file(file)?))
220        }
221    }
222
223    pub fn apply_overrides(&mut self, overrides: HashMap<String, Icon>) {
224        self.0.extend(overrides);
225    }
226
227    pub fn get(&self, icon: &'_ str, value: Option<f64>) -> Option<&str> {
228        match (self.0.get(icon)?, value) {
229            (Icon::Single(icon), _) => Some(icon),
230            (Icon::Progression(prog), _) if prog.is_empty() => None,
231            (Icon::Progression(prog), None) => Some(prog.last().unwrap()),
232            (Icon::Progression(prog), Some(value)) => {
233                let index = ((value * prog.len() as f64) as usize).clamp(0, prog.len() - 1);
234                Some(prog[index].as_str())
235            }
236        }
237    }
238}

240#[derive(Deserialize, Default)]
241#[serde(deny_unknown_fields, default)]
242struct IconsConfigRaw {
243    icons: Option<String>,
244    overrides: Option<HashMap<String, Icon>>,
245}
246
247impl TryFrom<IconsConfigRaw> for Icons {
248    type Error = Error;
249
250    fn try_from(raw: IconsConfigRaw) -> Result<Self, Self::Error> {
251        let mut icons = Self::from_file(raw.icons.as_deref().unwrap_or("none"))?;
252        if let Some(overrides) = raw.overrides {
253            for icon in overrides {
254                icons.0.insert(icon.0, icon.1);
255            }
256        }
257        Ok(icons)
258    }
259}

261#[cfg(test)]
262mod tests {
263    use super::*;
264
265    #[test]
266    fn default_icons_cover_every_key() {
267        let icons = Icons::default();
268        let mut defaults: Vec<&str> = icons.0.keys().map(String::as_str).collect();
269        let mut keys = ALL.to_vec();
270        defaults.sort_unstable();
271        keys.sort_unstable();
272        assert_eq!(defaults, keys);
273    }
