1//! Terminal images for doctor: lets a terminal that speaks the kitty graphics
2//! protocol show glyphs as the bar draws them, not as the terminal's own
3//! fonts would.
4//!
5//! Detection follows kitty's documentation: a graphics query, then a cell
6//! size request, then a Primary Device Attributes (DA) request. Every
7//! terminal answers DA, so DA is the sentinel and a terminal that ignores
8//! the rest never makes us wait for the timeout.
9//!
10//! ```text
11//!   us  ──► ESC _G i=31,a=q,… ESC \   ESC [ 16 t   ESC [ c
12//!   tty ◄── ESC _G i=31;OK ESC \       (only with kitty graphics)
13//!   tty ◄── ESC [ 6 ; 20 ; 10 t        (cell size, when supported)
14//!   tty ◄── ESC [ ? 62 ; 22 c          (always)
15//! ```
16

24/// A 1×1 query-only image (`a=q`), the cell size request, and DA.
25const QUERY: &[u8] = b"\x1b_Gi=31,s=1,v=1,a=q,t=d,f=24;AAAA\x1b\\\x1b[16t\x1b[c";
26const KITTY_OK: &[u8] = b"\x1b_Gi=31;OK";
27const DA_START: &[u8] = b"\x1b[?";
28const CELL_SIZE_START: &[u8] = b"\x1b[6;";
29const REPLY_TIMEOUT: Duration = Duration::from_millis(500);
30
31/// The protocol caps each escape's payload at 4096 bytes of base64.
32const CHUNK: usize = 4096;
33const PNG_SIGNATURE: &[u8] = b"\x89PNG\r\n\x1a\n";
34
35/// One terminal cell, in pixels.
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37pub struct CellSize {
38    pub width: u32,
39    pub height: u32,
40}
41
42/// The cell size of a terminal that can show kitty images on stdout, or
43/// `None` when doctor has to stay text-only: stdout is not a terminal, the
44/// terminal has no kitty graphics (alacritty, tmux without passthrough), or
45/// it does not report its cell size.
46pub fn kitty_cell_size() -> Option<CellSize> {
47    if !std::io::stdout().is_terminal() {
48        return None;
49    }
50    let reply = query_tty().ok()?;
51    find(&reply, KITTY_OK)?;
52    cell_size(&reply)
53}

55/// A PNG drawn one cell high where the cursor is, and how many cells wide it
56/// comes out. The cursor does not move (`C=1`): the caller pads with spaces,
57/// so table alignment never depends on the terminal moving it right.
58pub fn png_image(png: &[u8], cell: CellSize) -> Option<(String, usize)> {
59    let (width, height) = png_size(png)?;
60    let cells = cells_wide(width, height, cell);
61    Some((escape(png), cells))
62}
63
64/// The escapes that place `png`: one row high, no cursor movement, no
65/// replies (`q=2`, which would otherwise arrive as input). Payloads longer
66/// than [`CHUNK`] continue in `m=1` escapes.
67fn escape(png: &[u8]) -> String {
68    let payload = base64::engine::general_purpose::STANDARD.encode(png);
69    let chunks: Vec<&[u8]> = payload.as_bytes().chunks(CHUNK).collect();
70
71    let mut out = String::new();
72    for (i, chunk) in chunks.iter().enumerate() {
73        let more = u8::from(i + 1 < chunks.len());
74        let keys = if i == 0 {
75            format!("a=T,f=100,r=1,C=1,q=2,m={more}")
76        } else {
77            format!("m={more}")
78        };
79        // base64 is ASCII.
80        let chunk = std::str::from_utf8(chunk).unwrap_or_default();
81        out.push_str(&format!("\x1b_G{keys};{chunk}\x1b\\"));
82    }
83    out
84}

86/// Width and height from the PNG header (IHDR follows the signature).
87fn png_size(png: &[u8]) -> Option<(u32, u32)> {
88    if !png.starts_with(PNG_SIGNATURE) {
89        return None;
90    }
91    let dimension = |at: usize| Some(u32::from_be_bytes(png.get(at..at + 4)?.try_into().ok()?));
92    Some((dimension(16)?, dimension(20)?))
93}

95/// Cells an image covers once scaled to one cell high, rounded up.
96fn cells_wide(width: u32, height: u32, cell: CellSize) -> usize {
97    if height == 0 || cell.width == 0 {
98        return 1;
99    }
100    let scaled = u64::from(width) * u64::from(cell.height) / u64::from(height);
101    scaled.div_ceil(u64::from(cell.width)).max(1) as usize
102}

104/// Raw mode for the duration of the probe: no line buffering, no echo of the
105/// terminal's reply. Restores the previous mode when dropped.
106struct RawMode {
107    fd: i32,
108    saved: libc::termios,
109}
110
111impl RawMode {
112    fn enter(fd: i32) -> std::io::Result<Self> {
113        // SAFETY: tcgetattr only writes into the termios we pass it.
114        let mut saved: libc::termios = unsafe { std::mem::zeroed() };
115        if unsafe { libc::tcgetattr(fd, &mut saved) } != 0 {
116            return Err(std::io::Error::last_os_error());
117        }
118
119        let mut raw = saved;
120        raw.c_lflag &= !(libc::ICANON | libc::ECHO);
121        raw.c_cc[libc::VMIN] = 0;
122        raw.c_cc[libc::VTIME] = 0;
123        // SAFETY: fd is an open terminal and raw is a valid termios.
124        if unsafe { libc::tcsetattr(fd, libc::TCSANOW, &raw) } != 0 {
125            return Err(std::io::Error::last_os_error());
126        }
127        Ok(Self { fd, saved })
128    }
129}
130
131impl Drop for RawMode {
132    fn drop(&mut self) {
133        // SAFETY: restoring the termios read from this same fd.
134        unsafe { libc::tcsetattr(self.fd, libc::TCSANOW, &self.saved) };
135    }
136}

138/// Send [`QUERY`] to the controlling terminal and collect the reply up to
139/// the DA answer. Fails when there is no terminal (piped, cron, CI).
140fn query_tty() -> std::io::Result<Vec<u8>> {
141    let mut tty = OpenOptions::new().read(true).write(true).open("/dev/tty")?;
142    let fd = tty.as_raw_fd();
143    let _raw = RawMode::enter(fd)?;
144
145    tty.write_all(QUERY)?;
146    tty.flush()?;
147
148    // Read until the DA reply is complete, or give up at the deadline.
149    let deadline = Instant::now() + REPLY_TIMEOUT;
150    let mut reply = Vec::new();
151    while !da_complete(&reply) {
152        let left = deadline.saturating_duration_since(Instant::now());
153        if left.is_zero() || !readable(fd, left)? {
154            break;
155        }
156
157        let mut buf = [0u8; 256];
158        let n = tty.read(&mut buf)?;
159        if n == 0 {
160            break;
161        }
162        reply.extend_from_slice(&buf[..n]);
163    }
164    Ok(reply)
165}

167/// Whether `fd` has input within `timeout`.
168fn readable(fd: i32, timeout: Duration) -> std::io::Result<bool> {
169    let mut pfd = libc::pollfd {
170        fd,
171        events: libc::POLLIN,
172        revents: 0,
173    };
174    let ms = i32::try_from(timeout.as_millis()).unwrap_or(i32::MAX);
175    // SAFETY: one valid pollfd.
176    match unsafe { libc::poll(&mut pfd, 1, ms) } {
177        -1 => Err(std::io::Error::last_os_error()),
178        0 => Ok(false),
179        _ => Ok(true),
180    }
181}

183fn find(haystack: &[u8], needle: &[u8]) -> Option<usize> {
184    haystack.windows(needle.len()).position(|w| w == needle)
185}
186
187/// Whether the reply holds a full DA response: `ESC [ ? … c`.
188fn da_complete(reply: &[u8]) -> bool {
189    find(reply, DA_START).is_some_and(|start| reply[start..].contains(&b'c'))
190}
191
192/// The cell size reply: `ESC [ 6 ; height ; width t`.
193fn cell_size(reply: &[u8]) -> Option<CellSize> {
194    let start = find(reply, CELL_SIZE_START)? + CELL_SIZE_START.len();
195    let body = &reply[start..];
196    let end = body.iter().position(|&b| b == b't')?;
197
198    let text = std::str::from_utf8(&body[..end]).ok()?;
199    let (height, width) = text.split_once(';')?;
200    let cell = CellSize {
201        width: width.parse().ok()?,
202        height: height.parse().ok()?,
203    };
204    (cell.width > 0 && cell.height > 0).then_some(cell)
205}

207#[cfg(test)]
208mod tests {
209    use super::*;
210
211    const CELL: CellSize = CellSize {
212        width: 10,
213        height: 20,
214    };
215
216    /// A PNG header for a `width`×`height` image; only IHDR is read.
217    fn png_header(width: u32, height: u32) -> Vec<u8> {
218        let mut png = PNG_SIGNATURE.to_vec();
219        png.extend([0, 0, 0, 13]);
220        png.extend(b"IHDR");
221        png.extend(width.to_be_bytes());
222        png.extend(height.to_be_bytes());
223        png
224    }
225
226    #[test]
227    fn cell_size_is_read_from_the_reply() {
228        let reply = b"\x1b_Gi=31;OK\x1b\\\x1b[6;20;10t\x1b[?62;22c";
229        assert_eq!(cell_size(reply), Some(CELL));
230    }
231
232    #[test]
233    fn no_cell_size_without_the_reply() {
234        // e.g. a terminal that answers DA only
235        assert_eq!(cell_size(b"\x1b[?6c"), None);
236        assert_eq!(cell_size(b"\x1b[6;0;10t"), None);
237    }
238
239    #[test]
240    fn da_is_complete_only_once_terminated() {
241        assert!(!da_complete(b"\x1b[?62;4"));
242        assert!(da_complete(b"\x1b[?62;4c"));
243    }
244
245    #[test]
246    fn png_size_is_read_from_the_header() {
247        assert_eq!(png_size(&png_header(15, 17)), Some((15, 17)));
248        assert_eq!(png_size(b"GIF89a"), None);
249        assert_eq!(png_size(PNG_SIGNATURE), None);
250    }
251
252    #[test]
253    fn width_in_cells_follows_the_one_row_scale() {
254        // 15×15 at a 20px row is 20px wide: two 10px cells.
255        assert_eq!(cells_wide(15, 15, CELL), 2);
256        // 30×15 doubles: 40px, four cells.
257        assert_eq!(cells_wide(30, 15, CELL), 4);
258        // A sliver still takes a cell.
259        assert_eq!(cells_wide(1, 15, CELL), 1);
260    }
261
262    #[test]
263    fn image_is_placed_without_moving_the_cursor_or_replying() {
264        let (escape, cells) = png_image(&png_header(15, 15), CELL).unwrap();
265        assert_eq!(cells, 2);
266        assert!(escape.starts_with("\x1b_Ga=T,f=100,r=1,C=1,q=2,m=0;"));
267        assert!(escape.ends_with("\x1b\\"));
268    }
269
270    #[test]
271    fn long_payloads_continue_in_chunks() {
272        let png = vec![0u8; CHUNK]; // base64 grows it past one chunk
273        let escape = escape(&png);
274        let parts: Vec<&str> = escape.split("\x1b\\").filter(|p| !p.is_empty()).collect();
275        assert_eq!(parts.len(), 2);
276        assert!(parts[0].starts_with("\x1b_Ga=T,f=100,r=1,C=1,q=2,m=1;"));
277        assert!(parts[1].starts_with("\x1b_Gm=0;"));
278    }
