1//! Network information
2//!
3//! This block uses `sysfs` and `netlink` and thus does not require any external dependencies.
4//!
5//! # Configuration
6//!
7//! Key | Values | Default
8//! ----|--------|--------
9//! `device` | Network interface to monitor (as specified in `/sys/class/net/`). Supports regex. | If not set, device will be automatically selected every `interval`
10//! `interval` | Update interval in seconds | `2`
11//! `format` | A [MultiFormat][MaybeMultiFormatConfig] string to customise the output of this block. See below for available placeholders. | `[" $icon ^icon_net_down $speed_down.eng(prefix:K) ^icon_net_up $speed_up.eng(prefix:K) "]`
12//! `inactive_format` | Same as `format` but for when the interface is inactive | `" $icon Down "`
13//! `missing_format` | Same as `format` but for when the device is missing | `" × "`
14//!
15//! Action          | Description                               | Default button
16//! ----------------|-------------------------------------------|---------------
17//! `toggle_format` **DEPRECATED** | Toggles between `format` and `format_alt` | -
18//! `next_format`  | Switches to the next format in the list     | Left
19//! `prev_format`  | Switches to the previous format in the list | Right
20//!
21//! Placeholder       | Value                       | Type   | Unit
22//! ------------------|-----------------------------|--------|---------------
23//! `icon`            | Icon based on device's type | Icon   | -
24//! `speed_down`      | Download speed              | Number | Bytes per second
25//! `speed_up`        | Upload speed                | Number | Bytes per second
26//! `graph_down`      | Download speed graph        | Text   | -
27//! `graph_up`        | Upload speed graph          | Text   | -
28//! `device`          | The name of device          | Text   | -
29//! `ssid`            | Network SSID (WiFi only)    | Text   | -
30//! `frequency`       | WiFi frequency              | Number | Hz
31//! `signal_strength` | WiFi signal                 | Number | %
32//! `bitrate`         | WiFi connection bitrate     | Number | Bits per second
33//! `ip`              | IPv4 address of the iface   | Text   | -
34//! `ipv6`            | IPv6 address of the iface   | Text   | -
35//! `nameserver`      | Nameserver                  | Text   | -
36//!
37//! # Example
38//!
39//! Display WiFi info if available
40//!
41//! ```toml
42//! [[block]]
43//! block = "net"
44//! format = " $icon {$signal_strength $ssid $frequency|Wired connection} via $device "
45//! ```
46//!
47//! Display exact device
48//!
49//! ```toml
50//! [[block]]
51//! block = "net"
52//! device = "^wlo0$"
53//! ```
54//!
55//! # Icons Used
56//! - `net_loopback` (`$icon`)
57//! - `net_vpn` (`$icon`)
58//! - `net_wired` (`$icon`)
59//! - `net_wireless` (`$icon`, as a progression)
60//! - `net_up` (`^icon_net_up`)
61//! - `net_down` (`^icon_net_down`)

82pub(crate) fn prepare(config: &Config) -> Result<Arc<BlockPlan>> {
83    // Every state that renders a device sets these on every render; the
84    // wifi-only values (ssid, signal_strength, ...) and the addresses are
85    // conditional and stay undeclared.
86    let device_output =
87        |output: OutputPlan| output.icon("icon", IconChoices::fixed(NetDevice::ALL_ICONS));
88    let formats = config.formats.with_default(
89        " $icon ^icon_net_down $speed_down.eng(prefix:K) ^icon_net_up $speed_up.eng(prefix:K) ",
90    )?;
91    // One output per format the user can rotate to, plus the two states that
92    // are not format variants at all: an inactive interface and a missing
93    // device each have their own format and are selected by `run()`.
94    let mut outputs = format_outputs(formats, device_output);
95    outputs.push(device_output(OutputPlan::new(
96        "inactive",
97        config.inactive_format.with_default(" $icon Down ")?,
98    )));
99    // `missing` sets no values at all: nothing to declare.
100    outputs.push(OutputPlan::new(
101        "missing",
102        config.missing_format.with_default(" × ")?,
103    ));
104    BlockPlan::new(outputs)
105}

107pub(crate) async fn run(config: &Config, api: &CommonApi, plan: &Arc<BlockPlan>) -> Result<()> {
108    let mut actions = api.get_actions()?;
109    api.set_default_actions(&[
110        (MouseButton::Left, None, "next_format"),
111        (MouseButton::Right, None, "prev_format"),
112    ])?;
113
114    // The plan also holds the `inactive` and `missing` states; the rotation
115    // covers the format outputs alone.
116    let mut formats = FormatRotation::new(plan)?;
117    let output_inactive = plan.output("inactive")?;
118    let output_missing = plan.output("missing")?;
119
120    let mut timer = config.interval.timer();
121
122    let device_re = config
123        .device
124        .as_deref()
125        .map(Regex::new)
126        .transpose()
127        .error("Failed to parse device regex")?;
128
129    // Stats
130    let mut stats = None;
131    let mut stats_timer = Instant::now();
132    let mut tx_hist = [0f64; 8];
133    let mut rx_hist = [0f64; 8];
134
135    loop {
136        match NetDevice::new(device_re.as_ref()).await? {
137            None => {
138                api.set_widget(output_missing.new_widget())?;
139            }
140            Some(device) => {
141                let output = if device.is_up() {
142                    formats.current()
143                } else {
144                    &output_inactive
145                };
146                let mut widget = output.new_widget();
147
148                let mut speed_down: f64 = 0.0;
149                let mut speed_up: f64 = 0.0;
150
151                // Calculate speed
152                match (stats, device.iface.stats) {
153                    // No previous stats available
154                    (None, new_stats) => stats = new_stats,
155                    // No new stats available
156                    (Some(_), None) => stats = None,
157                    // All stats available
158                    (Some(old_stats), Some(new_stats)) => {
159                        let diff = new_stats - old_stats;
160                        let elapsed = stats_timer.elapsed().as_secs_f64();
161                        stats_timer = Instant::now();
162                        speed_down = diff.rx_bytes as f64 / elapsed;
163                        speed_up = diff.tx_bytes as f64 / elapsed;
164                        stats = Some(new_stats);
165                    }
166                }
167                push_to_hist(&mut rx_hist, speed_down);
168                push_to_hist(&mut tx_hist, speed_up);
169
170                let icon = if let Some(signal) = device.signal() {
171                    Value::icon_progression(device.icon, signal / 100.0)
172                } else {
173                    Value::icon(device.icon)
174                };
175
176                widget.set_values(map! {
177                    "icon" => icon,
178                    "speed_down" => Value::bytes(speed_down),
179                    "speed_up" => Value::bytes(speed_up),
180                    "graph_down" => Value::text(util::format_bar_graph(&rx_hist)),
181                    "graph_up" => Value::text(util::format_bar_graph(&tx_hist)),
182                    [if let Some(v) = device.ip] "ip" => Value::text(v.to_string()),
183                    [if let Some(v) = device.ipv6] "ipv6" => Value::text(v.to_string()),
184                    [if let Some(v) = device.ssid()] "ssid" => Value::text(v),
185                    [if let Some(v) = device.frequency()] "frequency" => Value::hertz(v),
186                    [if let Some(v) = device.bitrate()] "bitrate" => Value::bits(v),
187                    [if let Some(v) = device.signal()] "signal_strength" => Value::percents(v),
188                    [if !device.nameservers.is_empty()] "nameserver" => Value::text(
189                                                                            device
190                                                                                .nameservers
191                                                                                .into_iter()
192                                                                                .map(|s| s.to_string())
193                                                                                .join(" "),
194                                                                        ),
195                    "device" => Value::text(device.iface.name),
196                });
197
198                api.set_widget(widget)?;
199            }
200        }
201
202        loop {
203            select! {
204                _ = timer.tick() => break,
205                _ = api.wait_for_update_request() => break,
206                Some(action) = actions.recv() => match action.as_ref() {
207                    "next_format" | "toggle_format" => {
208                        formats.next();
209                        break;
210                    }
211                    "prev_format" => {
212                        formats.prev();
213                        break;
214                    }
215                    _ => ()
216                }
217            }
218        }
219    }
220}

222fn push_to_hist<T>(hist: &mut [T], elem: T) {
223    hist[0] = elem;
224    hist.rotate_left(1);
225}

231    #[test]
232    fn plan_declares_device_icon_set_on_every_rendering_state() {
233        let plan = prepare(&Config::default()).unwrap();
234        let ids: Vec<_> = plan.outputs().map(|o| o.id()).collect();
235        assert_eq!(ids, ["format", "inactive", "missing"]);
236
237        for id in ["format", "inactive"] {
238            let output = plan.output(id).unwrap();
239            let choices = output.output().choices_for("icon").unwrap();
240            for icon in NetDevice::ALL_ICONS {
241                assert!(choices.permits(icon), "{id} must permit {icon}");
242            }
243            assert!(!choices.permits("net_up"));
244        }
245
246        // `missing` renders a bare format and sets no values at all.
247        let missing = plan.output("missing").unwrap();
248        assert_eq!(missing.output().icon_placeholders().count(), 0);
249    }
250
251    #[test]
252    fn every_configured_format_becomes_an_output() {
253        let plan = prepare(&Config::default()).unwrap();
254        assert!(plan.output("format2").is_err());
255
256        let config: Config = toml::from_str(r#"format = [" $icon ", " $icon $device "]"#).unwrap();
257        let plan = prepare(&config).unwrap();
258        let ids: Vec<_> = plan.outputs().map(|o| o.id()).collect();
259        // The extra states stay alongside the per-format outputs.
260        assert_eq!(ids, ["format", "format2", "inactive", "missing"]);
261
262        let second = plan.output("format2").unwrap();
263        assert!(second.format().contains_key("device"));
264        let choices = second.output().choices_for("icon").unwrap();
265        assert!(choices.permits("net_wireless"));
266    }

268    #[test]
269    fn the_arrows_in_the_default_format_are_part_of_the_icon_surface() {
270        // They are `^icon_*` tokens rather than icon values, so they are not
271        // declared placeholders — but the block still draws them.
272        let plan = prepare(&Config::default()).unwrap();
273        let format = plan.output("format").unwrap();
274        assert_eq!(
275            format.output().static_icons(),
276            ["net_down", "net_up"],
277            "the default format's arrows must be visible to an inspector"
278        );
279        // The states that render no format icons say so.
280        for id in ["inactive", "missing"] {
281            let output = plan.output(id).unwrap();
282            assert!(output.output().static_icons().is_empty(), "{id}");
283        }
284    }

286    #[test]
287    fn every_chooser_icon_is_declared() {
288        // Exercise every branch of the runtime icon chooser and check the
289        // result against the declared set, so the two cannot drift apart.
290        let mut seen = Vec::new();
291        for is_wireless in [false, true] {
292            for tun_wg_ppp in [false, true] {
293                for name in ["lo", "eth0", "tun0", "wlan0"] {
294                    let icon = NetDevice::icon_for(is_wireless, tun_wg_ppp, name);
295                    assert!(
296                        NetDevice::ALL_ICONS.contains(&icon),
297                        "chooser returned undeclared icon {icon}"
298                    );
299                    if !seen.contains(&icon) {
300                        seen.push(icon);
301                    }
302                }
303            }
304        }
305        // ...and every declared name is actually reachable.
306        assert_eq!(seen.len(), NetDevice::ALL_ICONS.len());
307    }

321        push_to_hist(&mut hist, 2);
322        assert_eq!(&hist, &[3, 0, 10, 2]);
