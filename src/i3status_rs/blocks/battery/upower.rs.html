13const DISPLAY_DEVICE_PATH: ObjectPath =
14    ObjectPath::from_static_str_unchecked("/org/freedesktop/UPower/devices/DisplayDevice");
15
16struct DeviceConnection {
17    device_proxy: DeviceProxy<'static>,
18    changes: PropertiesChangedStream,
19}
20
21impl DeviceConnection {
22    async fn new(
23        dbus_conn: &Connection,
24        upower_proxy: &UPowerProxy<'_>,
25        device: &DeviceName,
26        expected_model: Option<&str>,
27    ) -> Result<Option<Self>> {
28        let device_proxy =
29            if device.exact().is_none_or(|d| d == "DisplayDevice") && expected_model.is_none() {
30                device_proxy(dbus_conn, DISPLAY_DEVICE_PATH).await?
31            } else {
32                let mut res = None;
33                for path in upower_proxy
34                    .enumerate_devices()
35                    .await
36                    .error("Failed to retrieve UPower devices")?
37                {
38                    let proxy = device_proxy(dbus_conn, path).await?;
39                    match device_matches(&proxy, device, expected_model).await {
40                        Ok(true) => {
41                            res = Some(proxy);
42                            break;
43                        }
44                        Ok(false) => (),
45                        // Most likely removed since it was listed
46                        Err(err) => {
47                            debug!("skipping {}: {err}", proxy.inner().path());
48                        }
49                    }
50                }
51                match res {
52                    Some(res) => res,
53                    None => return Ok(None),
54                }
55            };
56
57        let changes = PropertiesProxy::builder(dbus_conn)
58            .destination("org.freedesktop.UPower")
59            .unwrap()
60            .path(device_proxy.inner().path().to_owned())
61            .unwrap()
62            .build()
63            .await
64            .error("Failed to create PropertiesProxy")?
65            .receive_properties_changed()
66            .await
67            .error("Failed to create PropertiesChangedStream")?;
68
69        Ok(Some(DeviceConnection {
70            device_proxy,
71            changes,
72        }))
73    }
74}

76/// Whether a device listed by UPower is the configured one.
77async fn device_matches(
78    proxy: &DeviceProxy<'_>,
79    device: &DeviceName,
80    expected_model: Option<&str>,
81) -> zbus::Result<bool> {
82    // Filter by model if needed
83    if let Some(expected_model) = expected_model
84        && let Ok(device_model) = proxy.model().await
85        && expected_model != device_model
86    {
87        return Ok(false);
88    }
89    // Verify device type
90    // https://upower.freedesktop.org/docs/Device.html#Device:Type
91    // consider any peripheral, UPS and internal battery
92    if proxy.type_().await? == 1 {
93        return Ok(false);
94    }
95    Ok(device.matches(&proxy.native_path().await?))
96}

98/// The properties are read from UPower each time rather than from zbus' cache. The cache is
99/// filled once, when the proxy is built, and stays unusable if UPower could not be reached then.
100/// It is also kept up to date by a task of its own, which may not have applied a change yet when
101/// the block wakes up for it.
102async fn device_proxy(
103    dbus_conn: &Connection,
104    path: impl Into<ObjectPath<'static>>,
105) -> Result<DeviceProxy<'static>> {
106    DeviceProxy::builder(dbus_conn)
107        .path(path.into())
108        .unwrap()
109        .cache_properties(CacheProperties::No)
110        .build()
111        .await
112        .error("Failed to create DeviceProxy")
113}

115pub(super) struct Device {
116    dbus_conn: Connection,
117    upower_proxy: UPowerProxy<'static>,
118    device: DeviceName,
119    dev_model: Option<String>,
120    device_conn: Option<DeviceConnection>,
121    owner_changed_stream: OwnerChangedStream<'static>,
122    device_added_stream: DeviceAddedStream,
123    device_removed_stream: DeviceRemovedStream,
124}
125
126impl Device {
127    pub(super) async fn new(device: DeviceName, dev_model: Option<String>) -> Result<Self> {
128        let dbus_conn = new_system_dbus_connection().await?;
129
130        let upower_proxy = UPowerProxy::new(&dbus_conn)
131            .await
132            .error("Could not create UPowerProxy")?;
133
134        // Subscribe before looking for the device, so that no change in between is missed. UPower
135        // does not have to be running yet: the streams follow its bus name.
136        let (owner_changed_stream, device_added_stream, device_removed_stream) = try_join! {
137            upower_proxy.inner().receive_owner_changed(),
138            upower_proxy.receive_device_added(),
139            upower_proxy.receive_device_removed()
140        }
141        .error("Could not create signal stream")?;
142
143        let mut this = Self {
144            dbus_conn,
145            upower_proxy,
146            device,
147            dev_model,
148            device_conn: None,
149            owner_changed_stream,
150            device_added_stream,
151            device_removed_stream,
152        };
153        this.device_conn = this.find_device().await;
154        Ok(this)
155    }
156
157    /// Look for the configured device.
158    ///
159    /// UPower may stop or restart at any time, so failing to talk to it means that there is no
160    /// device for now. The block looks again once UPower reappears or reports a new device.
161    async fn find_device(&self) -> Option<DeviceConnection> {
162        DeviceConnection::new(
163            &self.dbus_conn,
164            &self.upower_proxy,
165            &self.device,
166            self.dev_model.as_deref(),
167        )
168        .await
169        .unwrap_or_else(|err| {
170            debug!("could not find the device: {err}");
171            None
172        })
173    }
174
175    /// UPower stopped, started or was replaced. Nothing from the previous instance is valid
176    /// anymore, including the paths of its devices.
177    async fn on_owner_changed(&mut self, running: bool) {
178        self.device_conn = if running {
179            debug!("UPower appeared on the bus");
180            self.find_device().await
181        } else {
182            debug!("UPower left the bus");
183            None
184        };
185    }
186}

188#[async_trait]
189impl BatteryDevice for Device {
190    async fn get_info(&mut self) -> Result<Option<BatteryInfo>> {
191        match &self.device_conn {
192            None => Ok(None),
193            Some(device_conn) => {
194                match try_join! {
195                    device_conn.device_proxy.is_present(),
196                    device_conn.device_proxy.percentage(),
197                    device_conn.device_proxy.energy_rate(),
198                    device_conn.device_proxy.state(),
199                    device_conn.device_proxy.time_to_full(),
200                    device_conn.device_proxy.time_to_empty(),
201                } {
202                    // The display device keeps existing without any battery behind it, and
203                    // hot-removable batteries are listed while their bay is empty.
204                    Err(_) | Ok((false, ..)) => Ok(None),
205                    Ok((true, capacity, power, state, time_to_full, time_to_empty)) => {
206                        let status = match state {
207                            1 => BatteryStatus::Charging,
208                            2 | 6 => BatteryStatus::Discharging,
209                            3 => BatteryStatus::Empty,
210                            4 => BatteryStatus::Full,
211                            5 => BatteryStatus::NotCharging,
212                            _ => BatteryStatus::Unknown,
213                        };
214
215                        let time_remaining = match status {
216                            BatteryStatus::Charging => Some(time_to_full as f64),
217                            BatteryStatus::Discharging => Some(time_to_empty as f64),
218                            _ => None,
219                        };
220
221                        Ok(Some(BatteryInfo {
222                            status,
223                            capacity,
224                            power: Some(power),
225                            time_remaining,
226                        }))
227                    }
228                }
229            }
230        }
231    }
232
233    async fn wait_for_change(&mut self) -> Result<()> {
234        // The streams only end when the connection to the bus is lost. Returning an error then
235        // restarts the block, instead of polling ended streams in a busy loop.
236        loop {
237            match &mut self.device_conn {
238                Some(device_conn) => select! {
239                    Some(_) = device_conn.changes.next() => break,
240                    Some(_) = self.device_added_stream.next() => (),
241                    Some(msg) = self.device_removed_stream.next() => {
242                        let args = msg.args().error("Failed to get the args")?;
243                        if args.device().as_ref() == device_conn.device_proxy.inner().path().as_ref() {
244                            // Another device may match as well
245                            self.device_conn = self.find_device().await;
246                            break;
247                        }
248                    },
249                    Some(owner) = self.owner_changed_stream.next() => {
250                        self.on_owner_changed(owner.is_some()).await;
251                        break;
252                    },
253                    else => return Err(Error::new("Lost the connection to the system bus")),
254                },
255                None => select! {
256                    Some(_) = self.device_removed_stream.next() => (),
257                    Some(_) = self.device_added_stream.next() => {
258                        if let Some(device_conn) = self.find_device().await {
259                            self.device_conn = Some(device_conn);
260                            break;
261                        }
262                    },
263                    Some(owner) = self.owner_changed_stream.next() => {
264                        self.on_owner_changed(owner.is_some()).await;
265                        break;
266                    },
267                    else => return Err(Error::new("Lost the connection to the system bus")),
268                },
269            }
270        }
271
272        Ok(())
273    }
274}
