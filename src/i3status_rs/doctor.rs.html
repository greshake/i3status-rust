1//! `--doctor`: diagnose configuration problems.
2//!
3//! The report is assessment-first: short status lines for what works, a
4//! per-block live test, a per-glyph icon table, and a final Problems list
5//! where every finding gets a one-line diagnosis and a matter-of-fact fix.
6//!
7//! Sections:
8//! - Header: which config file, icon set file and bar font are in use (the
9//!   bar font is auto-detected from a running i3/sway over IPC; `--font`
10//!   overrides it).
11//! - Blocks: every configured block is instantiated and run for one cycle
12//!   (unless `--doctor-skip-live`), showing its rendered output or a
13//!   diagnosed failure — including render errors, which are explained with
14//!   the placeholders the block actually provided. Note that this performs
15//!   real work: network requests, command execution, D-Bus calls.
16//! - Icons: one row per glyph with its codepoint and the font that will
17//!   actually draw it (via fontconfig). Rows drawn by fonts outside the
18//!   bar's font list are highlighted. "Used by" lists the blocks that can
19//!   request the icon.
20//!
21//! - Problems: numbered findings with concrete fixes.
22//!
23//! A block can only request icons its prepared contract declares — the
24//! block plan each block builds from its configuration, checked when the
25//! block publishes a widget — so doctor validates the whole declared
26//! contract and takes every declared icon as in use by that block. It does not try to prove
27//! which declarations a particular configuration or runtime state would
28//! reach: a declaration a format never renders is reported as undefined
29//! just the same, because the contract is what the block is allowed to ask
30//! for. Blocks that choose names at runtime (`custom`, `custom_dbus`)
31//! declare that openly and are reported as such.
32//!
33//! Doctor continues after recoverable failures: a missing tool, an
34//! unreadable section, a block that fails its live test are reported (with
35//! install or fix instructions where known) and the rest of the report
36//! still runs. It stops early only when there is nothing left to inspect —
37//! a configuration file that is missing, unreadable, or not valid TOML.

100#[derive(Clone, Copy, PartialEq, Eq)]
101pub enum IconDisplay {

108pub fn run(
109    config_arg: &str,
110    font_arg: Option<&str>,
111    skip_live: bool,
112    display: IconDisplay,
113) -> usize {
114    let style = Style::detect();
115    let mut problems: Vec<Problem> = Vec::new();
116
117    println!("i3status-rs doctor");
118    println!();
119
120    // === Config file ===
121    let (resolved, trace) = resolve_file(config_arg, None);
122    let config_path = match resolved {
123        Resolved::Found(path) => path,
124        Resolved::Missing => {
125            problems.push(Problem {
126                diagnosis: escape_control_multiline(&format!(
127                    "Configuration file {config_arg:?} not found. Searched:\n{trace}"
128                )),
129                fix: Some(
130                    "Create ~/.config/i3status-rust/config.toml, or pass a path: i3status-rs --doctor <path>"
131                        .into(),
132                ),
133            });
134            print_problems(&problems, false, &style);
135            return problems.len();
136        }
137        Resolved::Failed(err) => {
138            problems.push(Problem {
139                diagnosis: escape_control_multiline(&format!(
140                    "Configuration file {config_arg:?} cannot be resolved: {err}. The bar's own \
141                     lookup stops at this candidate, so it would not start either — and the \
142                     locations after it are never consulted. Searched:\n{trace}"
143                )),
144                fix: Some(
145                    "Repair or remove that path; a broken symlink or an unreadable directory is \
146                     the usual cause."
147                        .into(),
148                ),
149            });
150            print_problems(&problems, false, &style);
151            return problems.len();
152        }
153    };
154    println!(
155        "Config:   {}",
156        escape_control(&config_path.display().to_string())
157    );
158
159    let raw: Option<toml::Value> = match std::fs::read_to_string(&config_path) {
160        Err(err) => {
161            problems.push(Problem {
162                diagnosis: escape_control_multiline(&format!(
163                    "Cannot read {}: {err}",
164                    config_path.display()
165                )),
166                fix: None,
167            });
168            None
169        }
170        Ok(text) => match toml::from_str(&text) {
171            Ok(value) => Some(value),
172            Err(err) => {
173                problems.push(Problem {
174                    diagnosis: format!(
175                    "Configuration is not valid TOML:\n{}",
176                    escape_control_multiline(&err.to_string())
177                ),
178                    fix: Some(
179                        "Fix the syntax error above; nothing else can be checked until the file parses."
180                            .into(),
181                    ),
182                });
183                None
184            }
185        },
186    };
187    let Some(raw) = raw else {
188        print_problems(&problems, false, &style);
189        return problems.len();
190    };
191
192    // === Icon set ===
193    // Parser complaints already explained in specific terms, so the typed
194    // parse below does not report the same defect a second time.
195    let mut explained_parse_errors: Vec<String> = Vec::new();
196    let builtin = Icons::default().0;
197    let (set_name, overrides_value) = icons_config(&raw, &mut problems);
198    let (base_map, set_desc) = if set_name == "none" {
199        (builtin.clone(), "built-in text icons".to_string())
200    } else {
201        let (file, trace) = resolve_file(set_name, Some("icons"));
202        match file {
203            Resolved::Failed(err) => {
204                problems.push(Problem {
205                    diagnosis: escape_control_multiline(&format!(
206                        "Icon set {set_name:?} cannot be resolved: {err}. The bar's own lookup \
207                         stops at this candidate, so the locations after it are never \
208                         consulted. Searched:\n{trace}"
209                    )),
210                    fix: Some(
211                        "Repair or remove that path; a broken symlink or an unreadable \
212                         directory is the usual cause."
213                            .into(),
214                    ),
215                });
216                (
217                    builtin.clone(),
218                    "built-in text icons (fallback)".to_string(),
219                )
220            }
221            Resolved::Missing => {
222                // The typed parse fails on the same lookup; let this
223                // specific diagnosis stand in for it.
224                explained_parse_errors.push(format!("Icon set '{set_name}' not found"));
225                problems.push(Problem {
226                    diagnosis: escape_control_multiline(&format!(
227                        "Icon set {set_name:?} not found. Icon sets are plain TOML files looked \
228                         up on disk, not built into the binary. Searched:\n{trace}"
229                    )),
230                    fix: Some(escape_control(&format!(
231                        "Copy the set (e.g. files/icons/{}.toml from the source tree) into \
232                         ~/.config/i3status-rust/icons/, or use an absolute path in `icons = `.",
233                        escape_control(set_name)
234                    ))),
235                });
236                (
237                    builtin.clone(),
238                    "built-in text icons (fallback)".to_string(),
239                )
240            }
241            Resolved::Found(file) => {
242                match util::deserialize_toml_file::<HashMap<String, Icon>, _>(&file) {
243                    Err(err) => {
244                        explained_parse_errors.push(err.to_string().trim().to_string());
245                        problems.push(Problem {
246                        diagnosis: escape_control_multiline(&format!(
247                            "Icon set file {} cannot be parsed: {err}",
248                            file.display()
249                        )),
250                        fix: Some(
251                            "Fix the file, or remove `icons = ` to use the built-in text icons."
252                                .into(),
253                        ),
254                    });
255                        (
256                            builtin.clone(),
257                            "built-in text icons (fallback)".to_string(),
258                        )
259                    }
260                    Ok(map) => {
261                        let file = file.display().to_string();
262                        let desc = if set_name == file {
263                            format!("{file} ({} names)", map.len())
264                        } else {
265                            format!("{set_name} ({file}, {} names)", map.len())
266                        };
267                        (map, desc)
268                    }
269                }
270            }
271        }
272    };
273    println!("Icon set: {}", escape_control(&set_desc));
274
275    let global_overrides: HashMap<String, Icon> = match overrides_value {
276        None => HashMap::new(),
277        Some(value) => match value.clone().try_into() {
278            Ok(overrides) => overrides,
279            Err(err) => {
280                explained_parse_errors.push(err.to_string().trim().to_string());
281                problems.push(Problem {
282                    diagnosis: escape_control_multiline(&format!(
283                        "[icons.overrides] is not a valid icon table: {err}"
284                    )),
285                    fix: None,
286                });
287                HashMap::new()
288            }
289        },
290    };
291
292    // Per-instance icons_overrides tables, inspected here so a malformed
293    // one is explained before the typed parse reports the same complaint.
294    let raw_overrides = raw_block_overrides(&raw, &mut problems, &mut explained_parse_errors);
295
296    // === Bar font ===
297    let detected = if font_arg.is_none() {
298        detect_bar_font()
299    } else {
300        None
301    };
302    let font_pattern = font_arg.or(detected.as_ref().map(|d| d.font.as_str()));
303    // X core fonts (XLFD, "-misc-fixed-...") bypass fontconfig entirely, so
304    // the glyph analysis would be meaningless.
305    let is_xlfd = font_pattern.is_some_and(|f| f.starts_with('-'));
306    let mut font_check = if is_xlfd {
307        None
308    } else {
309        FontCheck::new(font_pattern)
310    };
311    // Findings that depend on the bar font are only authoritative when the
312    // font is known unambiguously: given via --font, or auto-detected without
313    // ambiguity. Otherwise they are reported as notes, not problems.
314    let font_authoritative =
315        font_arg.is_some() || (detected.as_ref().is_some_and(|d| d.note.is_none()));
316    // Environment limitations are notes, not problems: they say what doctor
317    // could not check, not that the user's configuration is wrong, and must
318    // not affect the exit code.
319    match (&font_check, font_arg, &detected) {
320        (None, ..) if is_xlfd => {
321            println!(
322                "Bar font: {} (X core font)",
323                escape_control(font_pattern.unwrap_or(""))
324            );
325            println!(
326                "   note: XLFD fonts bypass fontconfig, so doctor cannot analyze which font\n   \
327                 draws each glyph; the font check is skipped."
328            );
329        }
330        (None, ..) => {
331            println!("Bar font: (unchecked)");
332            println!(
333                "   note: `fc-match` or `fc-list` is not available, so doctor cannot tell\n   \
334                 which font will draw each icon glyph — the most common cause of wrong\n   \
335                 icons. Install the fontconfig utilities (package `fontconfig` on most\n   \
336                 distros) and re-run."
337            );
338        }
339        (Some(_), Some(font), _) => {
340            println!("Bar font: {} (from --font)", escape_control(font));
341        }
342        (Some(_), None, Some(d)) => {
343            println!(
344                "Bar font: {} (auto-detected via IPC, {})",
345                escape_control(&d.font),
346                escape_control(&d.bar_id)
347            );
348            if let Some(note) = &d.note {
349                println!("   note: {}", escape_control(note));
350            }
351        }
352        (Some(check), None, None) => {
353            println!(
354                "Bar font: fontconfig default ({})",
355                escape_control(&check.base_family)
356            );
357            println!(
358                "   note: no --font given and no running i3/sway answered over IPC; glyph\n   \
359                 providers below may not match your actual bar and are reported as notes,\n   \
360                 not problems. Re-run with the `font` directive from the bar {{ }} section\n   \
361                 of your i3/sway config: i3status-rs --doctor --font \"pango:...\""
362            );
363            if let Some(note) = stale_socket_note() {
364                println!("   note: {note}");
365            }
366        }
367    }
368    if display == IconDisplay::Glyphs {
369        println!(
370            "   note: icons below are drawn with this terminal's fonts, which may not be your\n   \
371             bar's; the Code and \"Effectively provided by\" columns describe the bar.\n   \
372             --noicons shows icon names instead."
373        );
374    }
375    // === Per-instance pango markup ===
376    // The bar's final output is pango markup, and markup can come from an
377    // icons_format, from literal text in any of the block's formats, or
378    // from the resolved icon value itself. Doctor does not emulate pango:
379    // it conservatively marks the affected icons' providers inconclusive.
380    let block_names = raw_block_names(&raw);
381    let labels = instance_labels(&block_names);
382    let global_markup = raw
383        .get("icons_format")
384        .and_then(|v| v.as_str())
385        .is_some_and(contains_markup);
386    let mut markup_labels: HashSet<String> = HashSet::new();
387    // The wrapper applied to every icon a block renders: its own
388    // icons_format if it sets one, the global one otherwise.
389    let global_icons_format = raw
390        .get("icons_format")
391        .and_then(|v| v.as_str())
392        .unwrap_or("{icon}");
393    let mut icons_formats: HashMap<String, String> = HashMap::new();
394    for (index, label) in labels.iter().enumerate() {
395        let table = raw
396            .get("block")
397            .and_then(|b| b.as_array())
398            .and_then(|b| b.get(index))
399            .and_then(|b| b.as_table());
400        let local_icons_format = table
401            .and_then(|t| t.get("icons_format"))
402            .and_then(|v| v.as_str())
403            .map(contains_markup);
404        // Every format the block can show, whichever naming it uses: a
405        // `format` list, `full_format`, `disconnected_format`, ...
406        let format_markup = table.is_some_and(|t| {
407            t.iter().any(|(key, value)| {
408                (key == "format" || key.starts_with("format_") || key.ends_with("_format"))
409                    && format_value_is_markup(value)
410            })
411        });
412        // Error output is rendered by the bar too, with the block's
413        // effective error formats: its own if it sets them, the global
414        // ones otherwise.
415        let error_markup = ["error_format", "error_fullscreen_format"]
416            .into_iter()
417            .any(|key| match table.and_then(|t| t.get(key)) {
418                Some(value) => format_value_is_markup(value),
419                None => raw.get(key).is_some_and(format_value_is_markup),
420            });
421        icons_formats.insert(
422            label.clone(),
423            table
424                .and_then(|t| t.get("icons_format"))
425                .and_then(|v| v.as_str())
426                .unwrap_or(global_icons_format)
427                .to_string(),
428        );
429        if local_icons_format.unwrap_or(global_markup) || format_markup || error_markup {
430            markup_labels.insert(label.clone());
431        }
432    }
433    if let Some(check) = &font_check {
434        for (family, resolved) in &check.families {
435            // Generic fontconfig aliases (monospace, sans-serif, ...) always
436            // resolve to some concrete family; that is normal, not a missing
437            // font.
438            if is_generic_family(family) {
439                continue;
440            }
441            if !family_present(family) {
442                let diagnosis = escape_control(&format!(
443                    "Font {family:?} is in the bar's font list but not installed{}.",
444                    resolved
445                        .as_ref()
446                        .map(|r| format!(" (fontconfig silently uses {r:?} in its place)"))
447                        .unwrap_or_default()
448                ));
449                if font_authoritative {
450                    problems.push(Problem {
451                        diagnosis,
452                        fix: Some(format!(
453                            "Install {family:?}, or remove it from the bar's font directive."
454                        )),
455                    });
456                } else {
457                    println!("   note: {diagnosis} (guessed font; not counted as a problem)");
458                }
459            }
460        }
461    }
462    println!();
463
464    // === Blocks (live test) ===
465    // Everything is attributed per block INSTANCE: two blocks of the same
466    // type get distinct labels ("time#1", "time#2") so per-instance
467    // icons_overrides are analyzed separately.
468    let mut used_now: BTreeMap<String, HashSet<String>> = BTreeMap::new();
469
470    let parsed: Option<Config> = match util::deserialize_toml_file::<Config, _>(&config_path) {
471        Ok(config) => Some(config),
472        Err(err) => {
473            // The same defect the icon-set/overrides inspection above
474            // already described in specific terms would fail this parse
475            // too; do not report one root cause twice.
476            // Skip only what a specific problem above already explained:
477            // the typed parse repeats the same parser complaint verbatim.
478            let text = err.to_string();
479            if !explained_parse_errors
480                .iter()
481                .any(|explained| text.contains(explained))
482            {
483                problems.push(Problem {
484                    diagnosis: escape_control_multiline(&format!(
485                        "Configuration does not validate: {text}"
486                    )),
487                    fix: None,
488                });
489            }
490            None
491        }
492    };
493
494    // Icon name -> the block instances whose contract declares it. Every
495    // declared icon counts as in use by that block.
496    let mut declared_by: BTreeMap<String, Vec<String>> = BTreeMap::new();
497
498    // Labels whose block configuration failed to deserialize: the static
499    // problem below already covers them, so the live report must not
500    // diagnose the same root cause again.
501    let mut prepared_errors: HashSet<String> = HashSet::new();
502
503    // Blocks that can request icon names doctor cannot enumerate.
504    let mut open_labels: BTreeSet<String> = BTreeSet::new();
505
506    for (index, label) in labels.iter().enumerate() {
507        let label = label.clone();
508
509        // The prepared contract of this block instance.
510        let plan = parsed
511            .as_ref()
512            .and_then(|config| config.blocks.get(index))
513            .and_then(|entry| match entry.config.plan() {
514                Ok(plan) => Some(plan),
515                Err(err) => {
516                    // The block's own configuration did not deserialize (the
517                    // bar would show a configuration error in its place).
518                    problems.push(Problem {
519                        diagnosis: format!(
520                            "Block `{label}` cannot be prepared from its configuration: {err}"
521                        ),
522                        fix: None,
523                    });
524                    prepared_errors.insert(label.clone());
525                    None
526                }
527            });
528
529        if let Some(plan) = &plan {
530            let analysis = analyze_plan(plan);
531            if analysis.open {
532                open_labels.insert(label.clone());
533            }
534            // Direct ^icon_* tokens in effective formats (including
535            // defaults) count the same as declared icons: in use.
536            for icon in &analysis.direct {
537                used_now
538                    .entry(icon.clone())
539                    .or_default()
540                    .insert(label.clone());
541            }
542            for icon in &analysis.required {
543                declared_by
544                    .entry(icon.clone())
545                    .or_default()
546                    .push(label.clone());
547            }
548        }
549        // Errors render through the shared error-widget plan with the
550        // block's effective error formats; whatever those formats reach
551        // (typically the conditional `refresh` restart icon) is a latent
552        // requirement of every block.
553        let error_rel = parsed
554            .as_ref()
555            .and_then(|config| config.blocks.get(index).map(|entry| (config, entry)))
556            .map(|(config, entry)| {
557                let plan = crate::block_plan::error_plan(
558                    entry
559                        .common
560                        .error_format
561                        .with_default_config(&config.error_format),
562                    entry
563                        .common
564                        .error_fullscreen_format
565                        .with_default_config(&config.error_fullscreen_format),
566                    // The restart button (and its refresh icon) can only
567                    // appear when retries are limited; configuration-error
568                    // blocks are never restartable.
569                    entry.common.max_retries.is_some()
570                        && !matches!(entry.config, crate::blocks::BlockConfig::Err(..)),
571                )
572                .expect("the error plan has statically unique ids");
573                analyze_plan(&plan)
574            });
575        if let Some(error_rel) = &error_rel {
576            for icon in &error_rel.required {
577                let users = declared_by.entry(icon.clone()).or_default();
578                if !users.contains(&label) {
579                    users.push(label.clone());
580                }
581            }
582        }
583    }
584
585    let mut live_ran = false;
586    // What each block that produced output actually rendered, with the
587    // strings its own icons expanded to, for the text-glyph check below.
588    let mut rendered_texts: Vec<(String, String, Vec<String>)> = Vec::new();
589    // (label, icon) pairs already diagnosed by a live render error, so the
590    // icon table does not report the same root cause twice
591    let mut live_reported: HashSet<(String, String)> = HashSet::new();
592    let icons_skip_reason = parsed
593        .is_none()
594        .then_some("configuration does not validate");
595    match (parsed, skip_live) {
596        (None, _) => println!("Blocks: skipped (configuration does not validate)\n"),
597        (Some(_), true) => println!("Blocks: skipped (--doctor-skip-live)\n"),
598        (Some(config), false) => {
599            live_ran = true;
600            println!(
601                "Blocks (each run for one cycle; performs real requests/commands, {}s timeout)",
602                LIVE_TIMEOUT.as_secs()
603            );
604            let mut reports = run_live(config_arg, config.blocks.len(), &mut problems);
605            if display == IconDisplay::Names {
606                reports.iter_mut().for_each(name_icons);
607            }
608            let num_w = format!("{}", reports.len()).len().max("#".len());
609            let name_w = column_width("Name", reports.iter().map(|r| r.name.as_str()));
610            let out_w = reports
611                .iter()
612                .map(|r| UnicodeWidthStr::width(output_cell(&r.verdict).0.as_str()))
613                .chain(["Output".len()])
614                .max()
615                .unwrap_or(0);
616            println!(
617                "{:>num_w$}  {}  {}  Icons/text glyphs used during this run",
618                "#",
619                pad_right("Name", name_w),
620                pad_right("Output", out_w)
621            );
622            for report in &reports {
623                let label = labels
624                    .get(report.index)
625                    .cloned()
626                    .unwrap_or_else(|| report.name.clone());
627                match &report.verdict {
628                    LiveVerdict::Rendered {
629                        check_text,
630                        icon_glyphs,
631                        ..
632                    } => {
633                        // The block's actual output contains pango markup
634                        // (from a format, a pango-str value, or an icon):
635                        // which font draws its icons is not verifiable.
636                        if contains_markup(check_text) {
637                            markup_labels.insert(label.clone());
638                        }
639                        rendered_texts.push((
640                            label.clone(),
641                            check_text.clone(),
642                            icon_glyphs.clone(),
643                        ));
644                    }
645                    // A block skipped by if_command keeps its declared icon
646                    // obligations: the command's result is a property of
647                    // this moment, not of the configuration, and the bar
648                    // will render the block whenever it succeeds.
649                    LiveVerdict::RenderError { error, .. } => {
650                        // The name is delimited by the message's own
651                        // wording, not by the first quote: icon names may
652                        // contain apostrophes.
653                        if let Some(icon) = error
654                            .split_once("Icon '")
655                            .and_then(|(_, rest)| rest.rsplit_once("' not found"))
656                            .map(|(icon, _)| icon)
657                        {
658                            live_reported.insert((label.clone(), icon.to_string()));
659                        }
660                        // Rendering failed, possibly on an icon. The
661                        // prepared contract already names exactly what can
662                        // render — including whatever the render failed on —
663                        // so the static analysis stays authoritative.
664                    }
665                    _ => (),
666                }
667                print_block_report(
668                    report,
669                    num_w,
670                    name_w,
671                    out_w,
672                    &style,
673                    &mut problems,
674                    &mut used_now,
675                    &prepared_errors,
676                    &label,
677                );
678            }
679            if reports.iter().any(|report| {
680                rendered_flags(&report.verdict, &report.flags)
681                    .next()
682                    .is_some()
683            }) {
684                println!("{FLAG_FOOTNOTE}");
685            }
686            println!();
687        }
688    }
689
690    // === Icons table ===
691    let block_overrides: Vec<(String, HashMap<String, Icon>)> = raw_overrides
692        .into_iter()
693        .map(|(index, overrides)| (labels[index].clone(), overrides))
694        .collect();
695    // === Text glyphs in live output ===
696    // Blocks emit text glyphs too (country flags, unit symbols, ...) whose
697    // rendering depends on installed fonts just like icons. Doctor asks
698    // fontconfig the same decidable questions about them: substituted
699    // glyphs are worth a note (the common, working case), and glyphs no
700    // installed font provides are a problem (they render as empty boxes).
701    // Characters are classified one by one and only then grouped into
702    // consecutive same-verdict runs, so a sequence that renders (a country
703    // flag) stays intact while an unavailable neighbour is reported alone.
704    //
705    // (label, provider) -> sequences that provider draws for that block.
706    let mut text_fallbacks: BTreeMap<(String, String), Vec<String>> = BTreeMap::new();
707    let mut text_missing: BTreeMap<String, Vec<String>> = BTreeMap::new();
708    if let Some(check) = font_check.as_mut() {
709        for (label, text, icon_glyphs) in &rendered_texts {
710            if markup_labels.contains(label) {
711                continue;
712            }
713            // Characters this block's own icons contributed already have
714            // rows in the icon table. Another icon in the set that happens
715            // to use the same character says nothing about this text.
716            let icon_chars: HashSet<char> = icon_glyphs.iter().flat_map(|g| g.chars()).collect();
717            let mut runs: Vec<(Option<String>, String)> = Vec::new();
718            // Only characters that are adjacent in the output belong to the
719            // same sequence; anything skipped in between ends it.
720            let mut adjacent = false;
721            for c in text.chars() {
722                if !is_drawn(c) || icon_chars.contains(&c) {
723                    adjacent = false;
724                    continue;
725                }
726                // None = no installed font provides it
727                let verdict = match check.check(c) {
728                    GlyphFont::Base | GlyphFont::Configured(_) => {
729                        adjacent = false;
730                        continue;
731                    }
732                    GlyphFont::Fallback(family) => Some(first_family(&family)),
733                    GlyphFont::Missing => None,
734                };
735                match runs.last_mut() {
736                    Some((last, run)) if adjacent && *last == verdict => run.push(c),
737                    _ => runs.push((verdict, c.to_string())),
738                }
739                adjacent = true;
740            }
741            for (verdict, run) in runs {
742                let entry = match verdict {
743                    Some(family) => text_fallbacks.entry((label.clone(), family)).or_default(),
744                    None => text_missing.entry(label.clone()).or_default(),
745                };
746                if !entry.contains(&run) {
747                    entry.push(run);
748                }
749            }
750        }
751    }
752
753    print_icon_table(IconTableInput {
754        base_map: &base_map,
755        global_overrides: &global_overrides,
756        block_overrides: &block_overrides,
757        used_now: &used_now,
758        declared_by: &declared_by,
759        live_reported: &live_reported,
760        icons_formats: &icons_formats,
761        markup_labels: &markup_labels,
762        font_check: &mut font_check,
763        font_authoritative,
764        style: &style,
765        problems: &mut problems,
766        skip_reason: icons_skip_reason,
767        display,
768    });
769
770    if !text_fallbacks.is_empty() {
771        println!("Text glyphs in live output drawn by fonts outside the bar's font list:");
772        for ((label, family), runs) in &text_fallbacks {
773            let shown: Vec<String> = runs
774                .iter()
775                .map(|run| format!("{} ({})", escape_control(run), codepoints(run)))
776                .collect();
777            println!(
778                "   {} — {} ({label})",
779                shown.join(", "),
780                escape_control(family)
781            );
782        }
783        println!(
784            "   Like * rows above, these depend on which fonts happen to be installed; \
785             not counted as problems."
786        );
787        println!();
788    }
789    for (label, runs) in &text_missing {
790        let listed: Vec<String> = runs
791            .iter()
792            .map(|run| format!("{} ({})", escape_control(run), codepoints(run)))
793            .collect();
794        problems.push(Problem {
795            diagnosis: format!(
796                "`{label}` output contains glyph(s) no installed font provides \
797                 ({}): they render as empty boxes.",
798                listed.join(", ")
799            ),
800            fix: Some("Install a font that contains these codepoints.".into()),
801        });
802    }
803
804    if !open_labels.is_empty() {
805        let listed = open_labels
806            .iter()
807            .map(String::as_str)
808            .collect::<Vec<_>>()
809            .join(", ");
810        println!(
811            "note: dynamic icon source in: {listed}. These blocks choose icon names at runtime,\n\
812             so doctor cannot enumerate them in advance — {}",
813            if live_ran {
814                "the names above are what this run observed, not the complete set."
815            } else {
816                "and the live block test did not run, so none were observed."
817            }
818        );
819        println!();
820    }
821
822    print_problems(&problems, live_ran, &style);
823    problems.len()
824}

826/// What the bar's own escaping (`crate::escape`) turns plain text into. It
827/// changes no font, so it is not markup.
828const PLAIN_TEXT_ESCAPES: [&str; 4] = ["&amp;", "&lt;", "&gt;", "&#39;"];
829
830/// Whether a string is pango markup: a tag, or a character reference like
831/// `&#x1F600;` (which pango resolves without any tag being present).
832/// `"a &amp; b"` is plain text the bar escaped; `"<b>a</b>"` is markup.
833fn contains_markup(s: &str) -> bool {
834    if s.contains('<') {
835        return true;
836    }
837    let unescaped = PLAIN_TEXT_ESCAPES
838        .iter()
839        .fold(s.to_string(), |acc, escape| acc.replace(escape, ""));
840    unescaped.contains('&')
841}
842
843/// Whether a format value (string or {full, short} table) is markup.
844fn format_value_is_markup(value: &toml::Value) -> bool {
845    match value {
846        toml::Value::String(s) => contains_markup(s),
847        toml::Value::Table(t) => t.values().any(|v| v.as_str().is_some_and(contains_markup)),
848        // A block can rotate through several formats, so a format value may
849        // be a list of any of the above.
850        toml::Value::Array(items) => items.iter().any(format_value_is_markup),
851        _ => false,
852    }
853}

1015/// Extra time the parent grants a worker beyond the block's own deadline
1016/// before SIGKILLing its process group.
1017const WORKER_GRACE: Duration = Duration::from_secs(3);
1018
1019/// Run every block in its own worker process (`--doctor-worker <index>`, this
1020/// same binary). Future cancellation cannot reap OS processes a block has
1021/// spawned (e.g. the custom block's command), so each worker gets its own
1022/// process group, which the parent SIGKILLs after collecting the result —
1023/// nothing can hang doctor or outlive it.
1024fn run_live(config_arg: &str, count: usize, problems: &mut Vec<Problem>) -> Vec<BlockReport> {
1025    let exe = match std::env::current_exe() {
1026        Ok(exe) => exe,
1027        Err(err) => {
1028            return vec![BlockReport {
1029                index: 0,
1030                name: "<worker>".into(),
1031                verdict: LiveVerdict::BlockError(format!("cannot find own executable: {err}")),
1032                flags: Vec::new(),
1033            }];
1034        }
1035    };
1036    let runtime = match tokio::runtime::Builder::new_current_thread()
1037        .enable_all()
1038        .build()
1039    {
1040        Ok(runtime) => runtime,
1041        Err(err) => {
1042            return vec![BlockReport {
1043                index: 0,
1044                name: "<runtime>".into(),
1045                verdict: LiveVerdict::BlockError(format!("failed to start async runtime: {err}")),
1046                flags: Vec::new(),
1047            }];
1048        }
1049    };
1050
1051    // A block's subprocess can leave its worker's process group (setsid), so
1052    // the group SIGKILL alone cannot guarantee cleanup. Registering as a
1053    // child subreaper makes every orphaned descendant reparent to us instead
1054    // of init; after the workers are done we sweep and kill whatever is left.
1055    // SAFETY: plain prctl syscall
1056    let subreaper_ok = unsafe { libc::prctl(libc::PR_SET_CHILD_SUBREAPER, 1) } == 0;
1057    if !subreaper_ok {
1058        problems.push(Problem {
1059            diagnosis: "Cannot register as a child subreaper; processes spawned by blocks \
1060                        that detach from their process group may outlive doctor."
1061                .into(),
1062            fix: None,
1063        });
1064    }
1065
1066    let reports = runtime.block_on(async {
1067        let done = Arc::new(std::sync::atomic::AtomicUsize::new(0));
1068        let workers = (0..count).map(|index| {
1069            let exe = exe.clone();
1070            let done = done.clone();
1071            async move {
1072                let report = run_worker_process(&exe, config_arg, index).await;
1073                done.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
1074                report
1075            }
1076        });
1077        let work = futures::future::join_all(workers);
1078        // Blocks with slow commands or network requests can take their full
1079        // deadline; show progress while the user waits (tty only, so piped
1080        // output stays clean).
1081        if std::io::stdout().is_terminal() {
1082            use std::io::Write as _;
1083            const FRAMES: [char; 10] = ['⠋', '⠙', '⠹', '⠸', '⠼', '⠴', '⠦', '⠧', '⠇', '⠏'];
1084            let mut ticker = tokio::time::interval(Duration::from_millis(120));
1085            let mut frame = 0usize;
1086            tokio::pin!(work);
1087            loop {
1088                tokio::select! {
1089                    reports = &mut work => {
1090                        print!("\r\x1b[K");
1091                        let _ = std::io::stdout().flush();
1092                        break reports;
1093                    }
1094                    _ = ticker.tick() => {
1095                        print!(
1096                            "\r{} testing blocks… {}/{count} done",
1097                            FRAMES[frame % FRAMES.len()],
1098                            done.load(std::sync::atomic::Ordering::Relaxed),
1099                        );
1100                        let _ = std::io::stdout().flush();
1101                        frame += 1;
1102                    }
1103                }
1104            }
1105        } else {
1106            work.await
1107        }
1108    });
1109
1110    if subreaper_ok && let Err(leftover) = sweep_orphaned_children() {
1111        problems.push(Problem {
1112            diagnosis: format!(
1113                "Could not clean up all processes spawned by the block test: {leftover}"
1114            ),
1115            fix: Some("Check for leftover processes and kill them manually.".into()),
1116        });
1117    }
1118    reports
1119}

1121/// Kill every remaining child of this process. All legitimate children (the
1122/// workers) have already been reaped by this point, so anything left is an
1123/// escaped descendant of some block's subprocess tree, reparented to us by
1124/// the subreaper registration.
1125///
1126/// Each pass can uncover one more level of a nested chain (killing a parent
1127/// orphans its children, which then reparent to us), so this sweeps until a
1128/// pass finds nothing, bounded by a deadline rather than a pass count.
1129/// Returns Err with a description if processes remain at the deadline.
1130fn sweep_orphaned_children() -> Result<(), String> {
1131    let self_pid = std::process::id();
1132    let deadline = std::time::Instant::now() + Duration::from_secs(5);
1133    loop {
1134        // Reap any zombies first so they neither linger nor hide behind the
1135        // early return below.
1136        // SAFETY: plain syscall
1137        while unsafe { libc::waitpid(-1, std::ptr::null_mut(), libc::WNOHANG) } > 0 {}
1138        let mut found = 0usize;
1139        let dir = match std::fs::read_dir("/proc") {
1140            Ok(dir) => dir,
1141            Err(err) => return Err(format!("cannot scan /proc: {err}")),
1142        };
1143        for entry in dir.flatten() {
1144            let Some(pid) = entry
1145                .file_name()
1146                .to_str()
1147                .and_then(|s| s.parse::<u32>().ok())
1148            else {
1149                continue;
1150            };
1151            // The process may exit between the scan and the read; that is
1152            // fine, it is gone either way.
1153            let Ok(stat) = std::fs::read_to_string(format!("/proc/{pid}/stat")) else {
1154                continue;
1155            };
1156            // ppid is the second field after the parenthesized comm
1157            let Some((_, rest)) = stat.rsplit_once(')') else {
1158                continue;
1159            };
1160            let mut fields = rest.split_whitespace();
1161            let state = fields.next();
1162            let Some(ppid) = fields.next().and_then(|p| p.parse::<u32>().ok()) else {
1163                continue;
1164            };
1165            if ppid == self_pid && state != Some("Z") {
1166                // SAFETY: plain syscall. "No such process" (already gone) is
1167                // fine; any other failure leaves the process for the next
1168                // pass or the deadline report.
1169                unsafe { libc::kill(pid as i32, libc::SIGKILL) };
1170                found += 1;
1171            }
1172        }
1173        if found == 0 {
1174            return Ok(());
1175        }
1176        if std::time::Instant::now() > deadline {
1177            return Err(format!(
1178                "{found} process(es) still alive after 5s of sweeping"
1179            ));
1180        }
1181        // Give the kills a moment: children of the killed processes reparent
1182        // to us and are caught by the next iteration.
1183        std::thread::sleep(Duration::from_millis(10));
1184        // Reap the zombies so they vanish from the next scan.
1185        // SAFETY: plain syscall
1186        while unsafe { libc::waitpid(-1, std::ptr::null_mut(), libc::WNOHANG) } > 0 {}
1187    }
1188}

1190async fn run_worker_process(exe: &Path, config_arg: &str, index: usize) -> BlockReport {
1191    let fail = |msg: String| BlockReport {
1192        index,
1193        name: format!("block #{}", index + 1),
1194        verdict: LiveVerdict::BlockError(msg),
1195        flags: Vec::new(),
1196    };
1197
1198    let mut command = tokio::process::Command::new(exe);
1199    command
1200        .arg("--doctor-worker")
1201        .arg(index.to_string())
1202        // The parent may have been given the name after its own `--`;
1203        // without one here the worker reads a leading dash as an option.
1204        .arg("--")
1205        .arg(config_arg)
1206        .stdin(std::process::Stdio::null())
1207        .stdout(std::process::Stdio::piped())
1208        .stderr(std::process::Stdio::null())
1209        .process_group(0);
1210    let child = match command.spawn() {
1211        Ok(child) => child,
1212        Err(err) => return fail(format!("cannot start doctor worker: {err}")),
1213    };
1214    let pid = child.id();
1215
1216    let result = tokio::time::timeout(LIVE_TIMEOUT + WORKER_GRACE, child.wait_with_output()).await;
1217
1218    // Sweep the worker's whole process group unconditionally: this reaps any
1219    // subprocess a block spawned and left behind (the worker itself is
1220    // already gone in the normal case).
1221    if let Some(pid) = pid {
1222        // SAFETY: plain syscall; negative pid targets the process group
1223        unsafe { libc::kill(-(pid as i32), libc::SIGKILL) };
1224    }
1225
1226    match result {
1227        Err(_) => {
1228            // reap the killed worker so it does not linger as a zombie
1229            if let Some(pid) = pid {
1230                // SAFETY: plain syscall; the group was just SIGKILLed
1231                unsafe { libc::waitpid(pid as i32, std::ptr::null_mut(), 0) };
1232            }
1233            fail(format!(
1234                "doctor worker did not finish within {}s and was killed",
1235                (LIVE_TIMEOUT + WORKER_GRACE).as_secs()
1236            ))
1237        }
1238        Ok(Err(err)) => fail(format!("doctor worker failed: {err}")),
1239        Ok(Ok(output)) => match serde_json::from_slice::<BlockReport>(&output.stdout) {
1240            Ok(report) => report,
1241            Err(err) => fail(format!(
1242                "doctor worker produced no readable result ({err}); it may have crashed"
1243            )),
1244        },
1245    }
1246}

1248/// Entry point for `--doctor-worker <index>`: run one block's live test in
1249/// this process and print the result as JSON on stdout.
1250pub fn run_worker(config_arg: &str, index: usize) {
1251    // Workers are separate concurrent processes; give custom_dbus a unique
1252    // well-known name via an in-process channel — the environment every
1253    // block command and if_command sees stays exactly the user's own.
1254    // Unique per worker AND per doctor process: two doctor runs (or a
1255    // doctor run beside another) must not fight over the same name.
1256    crate::blocks::custom_dbus::set_doctor_dbus_suffix(format!(
1257        "doctor{}_{index}",
1258        std::process::id()
1259    ));
1260    util::enable_flag_recorder();
1261    let mut report = worker_report(config_arg, index);
1262    report.flags = util::recorded_flags();

1264    match serde_json::to_string(&report) {
1265        Ok(json) => println!("{json}"),
1266        Err(err) => eprintln!("doctor worker: cannot serialize report: {err}"),
1267    }
1268}

1270fn worker_report(config_arg: &str, index: usize) -> BlockReport {
1271    let fail = |msg: String| BlockReport {
1272        index,
1273        name: format!("block #{}", index + 1),
1274        verdict: LiveVerdict::BlockError(msg),
1275        flags: Vec::new(),
1276    };
1277    let Ok(Some(config_path)) = util::find_file(config_arg, None, Some("toml")) else {
1278        return fail("worker cannot find the configuration file".into());
1279    };
1280    let mut config: Config = match util::deserialize_toml_file(&config_path) {
1281        Ok(config) => config,
1282        Err(err) => return fail(format!("worker cannot parse the configuration: {err}")),
1283    };
1284    if index >= config.blocks.len() {
1285        return fail("worker got an out-of-range block index".into());
1286    }
1287    let entry = config.blocks.swap_remove(index);
1288    let shared = config.shared.clone();
1289    let geolocator = config.geolocator.clone();
1290
1291    let runtime = match tokio::runtime::Builder::new_current_thread()
1292        .enable_all()
1293        .build()
1294    {
1295        Ok(runtime) => runtime,
1296        Err(err) => return fail(format!("worker failed to start async runtime: {err}")),
1297    };
1298    let local = tokio::task::LocalSet::new();
1299    local.block_on(&runtime, test_block(index, entry, shared, geolocator))
1300}

1302async fn test_block(
1303    index: usize,
1304    entry: BlockConfigEntry,
1305    mut shared: SharedConfig,
1306    geolocator: Arc<Geolocator>,
1307) -> BlockReport {
1308    let name = entry.config.name().to_string();
1309    // One deadline shared by everything the block test does: if_command and
1310    // the block's first output together get LIVE_TIMEOUT, not each.
1311    let deadline = tokio::time::Instant::now() + LIVE_TIMEOUT;
1312
1313    if let Some(cmd) = &entry.common.if_command {
1314        // Subprocess cleanup is guaranteed by the worker's process group,
1315        // which the parent sweeps.
1316        let output = tokio::time::timeout_at(
1317            deadline,
1318            tokio::process::Command::new("sh")
1319                .args(["-c", cmd])
1320                .stdin(std::process::Stdio::null())
1321                .kill_on_drop(true)
1322                .output(),
1323        )
1324        .await;
1325        match output {
1326            Err(_) => {
1327                return BlockReport {
1328                    index,
1329                    name,
1330                    verdict: LiveVerdict::IfCommandTimeout(format!(
1331                        "if_command did not finish within {}s ({cmd})",
1332                        LIVE_TIMEOUT.as_secs()
1333                    )),
1334                    flags: Vec::new(),
1335                };
1336            }
1337            Ok(Err(err)) => {
1338                return BlockReport {
1339                    index,
1340                    name,
1341                    verdict: LiveVerdict::IfCommandFailed(format!(
1342                        "if_command could not run: {err}"
1343                    )),
1344                    flags: Vec::new(),
1345                };
1346            }
1347            Ok(Ok(output)) if !output.status.success() => {
1348                return BlockReport {
1349                    index,
1350                    name,
1351                    verdict: LiveVerdict::Skipped(format!(
1352                        "if_command exited non-zero ({cmd}) — the bar would not show this block"
1353                    )),
1354                    flags: Vec::new(),
1355                };
1356            }
1357            Ok(Ok(_)) => (),
1358        }
1359    }
1360
1361    // Apply per-block overrides the same way BarState::spawn_block does
1362    if let Some(icons_format) = entry.common.icons_format {
1363        shared.icons_format = Arc::new(icons_format);
1364    }
1365    if let Some(theme_overrides) = entry.common.theme_overrides
1366        && let Err(err) = Arc::make_mut(&mut shared.theme).apply_overrides(theme_overrides)
1367    {
1368        return BlockReport {
1369            index,
1370            name,
1371            verdict: LiveVerdict::BlockError(format!("invalid theme_overrides: {err}")),
1372            flags: Vec::new(),
1373        };
1374    }
1375    if let Some(icons_overrides) = entry.common.icons_overrides {
1376        Arc::make_mut(&mut shared.icons).apply_overrides(icons_overrides);
1377    }
1378
1379    let (sender, mut receiver) = mpsc::unbounded_channel();
1380    let api = CommonApi {
1381        id: index,
1382        update_request: Arc::new(tokio::sync::Notify::new()),
1383        request_sender: sender,
1384        // effectively "never": doctor only watches the first seconds of output
1385        error_interval: Duration::from_secs(3600),
1386        geolocator,
1387        max_retries: Some(0),
1388    };
1389
1390    let mut futures = futures::stream::FuturesUnordered::new();
1391    entry.config.spawn(api, &mut futures);
1392    let handle = tokio::task::spawn_local(async move {
1393        futures.next().await;
1394    });
1395
1396    let first = capture_output(&mut receiver, deadline).await;
1397
1398    handle.abort();
1399    if let Err(err) = handle.await
1400        && err.is_panic()
1401    {
1402        return BlockReport {
1403            index,
1404            name,
1405            verdict: LiveVerdict::Panicked(join_error_message(err)),
1406            flags: Vec::new(),
1407        };
1408    }
1409
1410    let verdict = match first {
1411        Err(_) => LiveVerdict::NoOutput,
1412        Ok(None) => LiveVerdict::Finished,
1413        Ok(Some(LiveOutput::Hidden)) => LiveVerdict::Hidden,
1414        Ok(Some(LiveOutput::Error(error))) => LiveVerdict::BlockError(unwrap_terminated(&error)),
1415        Ok(Some(LiveOutput::Widget(widget))) => render_widget(&widget, &shared, index),
1416    };
1417
1418    BlockReport {
1419        index,
1420        name,
1421        verdict,
1422        flags: Vec::new(),
1423    }
1424}

1426/// The block's output as the live test reports it: the latest thing it
1427/// publishes within `LIVE_SETTLE` of its first output. Blocks such as sound
1428/// publish before their backend reports the real state, so the first output
1429/// is not what the bar settles on. `Err` when nothing arrives by `deadline`.
1430async fn capture_output(
1431    receiver: &mut mpsc::UnboundedReceiver<Request>,
1432    deadline: tokio::time::Instant,
1433) -> std::result::Result<Option<LiveOutput>, tokio::time::error::Elapsed> {
1434    let Some(mut latest) = tokio::time::timeout_at(deadline, next_output(receiver)).await? else {
1435        return Ok(None);
1436    };
1437
1438    let settled = (tokio::time::Instant::now() + LIVE_SETTLE).min(deadline);
1439    while let Ok(Some(output)) = tokio::time::timeout_at(settled, next_output(receiver)).await {
1440        latest = output;
1441    }
1442    Ok(Some(latest))
1443}
1444
1445/// The next request that changes what the block shows; `None` once the
1446/// block's channel closes.
1447async fn next_output(receiver: &mut mpsc::UnboundedReceiver<Request>) -> Option<LiveOutput> {
1448    loop {
1449        let Request { cmd, .. } = receiver.recv().await?;
1450        match cmd {
1451            RequestCmd::SetWidget(widget) => return Some(LiveOutput::Widget(Box::new(widget))),
1452            RequestCmd::SetError { error, .. } => return Some(LiveOutput::Error(error)),
1453            RequestCmd::UnsetWidget => return Some(LiveOutput::Hidden),
1454            _ => continue,
1455        }
1456    }
1457}

1459fn render_widget(widget: &Widget, shared: &SharedConfig, index: usize) -> LiveVerdict {
1460    // Record which icons rendering actually consumes: syntactic inspection
1461    // of the format is not enough, because a fallback branch containing
1462    // $icon may never be evaluated.
1463    crate::config::enable_icon_recorder();
1464    let _ = crate::config::take_recorded_icons();
1465    match widget.get_data(shared, index) {
1466        Ok(segments) => {
1467            let recorded = crate::config::take_recorded_icons();
1468            let icon_strings = recorded.clone();
1469            let mut icons: Vec<String> = recorded.iter().map(|(name, _)| name.clone()).collect();
1470            icons.sort_unstable();
1471            icons.dedup();
1472            let mut icon_glyphs: Vec<String> =
1473                recorded.into_iter().map(|(_, glyph)| glyph).collect();
1474            icon_glyphs.sort_unstable();
1475            icon_glyphs.dedup();
1476            // i3bar needs a non-empty placeholder in the other field to
1477            // switch a block between full and short mode; it draws nothing
1478            // and is not part of what the block rendered.
1479            let real = |s: &str| (s != SHORT_MODE_SENTINEL).then(|| s.to_string());
1480            let full: String = segments
1481                .iter()
1482                .filter_map(|s| real(&s.full_text))
1483                .collect::<Vec<_>>()
1484                .join("");
1485            let short: String = segments
1486                .iter()
1487                .filter_map(|s| real(&s.short_text))
1488                .collect::<Vec<_>>()
1489                .join("");
1490            LiveVerdict::Rendered {
1491                check_text: format!("{full}{short}"),
1492                text: full,
1493                icon_glyphs,
1494                icons,
1495                icon_strings,
1496                provided_icons: provided_icon_values(widget),
1497            }
1498        }
1499        Err(error) => {
1500            let mut provided: Vec<String> = widget.values().keys().map(|k| k.to_string()).collect();
1501            provided.sort_unstable();
1502            LiveVerdict::RenderError {
1503                error: error.to_string(),
1504                provided,
1505                // rendering failed, possibly on an icon: count all of them
1506                icons: provided_icon_values(widget)
1507                    .into_iter()
1508                    .map(|(_, name)| name)
1509                    .collect(),
1510            }
1511        }
1512    }
1513}

1550/// Replace the strings a block's icons drew with `{name}`: `" 🔊 50% "`
1551/// becomes `" {volume} 50% "`. Longest first, so an icon string containing
1552/// another is replaced whole.
1553fn name_icons(report: &mut BlockReport) {
1554    let LiveVerdict::Rendered {
1555        text, icon_strings, ..
1556    } = &mut report.verdict
1557    else {
1558        return;
1559    };
1560    let mut strings: Vec<&(String, String)> = icon_strings
1561        .iter()
1562        .filter(|(_, drawn)| !drawn.is_empty())
1563        .collect();
1564    strings.sort_by_key(|(_, drawn)| std::cmp::Reverse(drawn.len()));
1565    for (name, drawn) in strings {
1566        // Keep the padding an icons_format adds around the glyph.
1567        let glyph = drawn.trim();
1568        let start = drawn.len() - drawn.trim_start().len();
1569        let named = format!(
1570            "{}{{{name}}}{}",
1571            &drawn[..start],
1572            &drawn[start + glyph.len()..]
1573        );
1574        *text = text.replace(drawn.as_str(), &named);
1575    }
1576}

1652/// Render control characters visibly, leaving printable text (including
1653/// icon glyphs and flags) untouched.
1654fn escape_control(text: &str) -> String {
1655    text.chars()
1656        .flat_map(|c| {
1657            // Invisible characters are shown, not obeyed: a bidi override
1658            // can reorder a whole line of doctor's own report. Visible text
1659            // is left alone, combining marks included.
1660            if c.is_control() {
1661                // \n, \t and friends read better than their codepoints.
1662                c.escape_debug().collect::<Vec<_>>()
1663            } else if is_invisible(c) {
1664                // Rust considers several of these printable, so the escape
1665                // has to be spelled out rather than delegated.
1666                format!("\\u{{{:x}}}", c as u32).chars().collect::<Vec<_>>()
1667            } else {
1668                vec![c]
1669            }
1670        })
1671        .collect()
1672}

1674/// Print one row of the live-test table and record its consequences
1675/// (problems, live icon usage).
1676#[allow(clippy::too_many_arguments)]
1677fn print_block_report(
1678    report: &BlockReport,
1679    num_w: usize,
1680    name_w: usize,
1681    out_w: usize,
1682    style: &Style,
1683    problems: &mut Vec<Problem>,
1684    used_now: &mut BTreeMap<String, HashSet<String>>,
1685    prepared_errors: &HashSet<String>,
1686    label: &str,
1687) {
1688    let (cell, red) = output_cell(&report.verdict);
1689    let icons: &[String] = match &report.verdict {
1690        LiveVerdict::Rendered { icons, .. } | LiveVerdict::RenderError { icons, .. } => icons,
1691        _ => &[],
1692    };
1693    // Icons by name, plus text glyphs the block generated at the source
1694    // (country flags): both are font-dependent output. Icon names can come
1695    // straight from a command's JSON, so escape them too.
1696    let mut cells: Vec<String> = icons.iter().map(|name| escape_control(name)).collect();
1697    cells.extend(
1698        rendered_flags(&report.verdict, &report.flags).map(|(code, flag)| flag_cell(code, flag)),
1699    );
1700    let icons_cell = if cells.is_empty() {
1701        "-".to_string()
1702    } else {
1703        cells.join(", ")
1704    };
1705    let line = format!(
1706        "{:>num_w$}  {}  {}  {icons_cell}",
1707        report.index + 1,
1708        pad_right(&report.name, name_w),
1709        pad_right(&cell, out_w)
1710    );
1711    if red {
1712        println!("{}{line}{}", style.red, style.reset);
1713    } else {
1714        println!("{line}");
1715    }
1716
1717    for icon in icons {
1718        used_now
1719            .entry(icon.clone())
1720            .or_default()
1721            .insert(label.to_string());
1722    }
1723
1724    match &report.verdict {
1725        LiveVerdict::RenderError {
1726            error, provided, ..
1727        } => {
1728            println!(
1729                "    values the block provided: {}",
1730                escape_control(&provided.join(", "))
1731            );
1732            let fix = if error.contains("Placeholder") {
1733                Some(
1734                    "The format references a placeholder the block did not provide (it may be \
1735                     state-dependent). Make it optional with a fallback: \"{ $placeholder |}\"."
1736                        .to_string(),
1737                )
1738            } else if error.contains("Icon") {
1739                Some(
1740                    "Add the icon name to [icons.overrides] or use an icon set that defines it \
1741                     (see the icon table below)."
1742                        .to_string(),
1743                )
1744            } else {
1745                None
1746            };
1747            problems.push(Problem {
1748                diagnosis: format!("{}: render error: {}", report.name, escape_control(error)),
1749                fix,
1750            });
1751        }
1752        LiveVerdict::BlockError(error) if error.contains(CONTRACT_BUG) => {
1753            // set_widget rejects a widget whose icons its plan does not
1754            // declare, so the violation arrives here as a block error.
1755            // That is a fault in i3status-rs, not in the configuration.
1756            problems.push(Problem {
1757                diagnosis: format!(
1758                    "`{}` violated its own output contract: {}. This is a bug in i3status-rs, \
1759                     not in your configuration.",
1760                    report.name,
1761                    escape_control(error)
1762                ),
1763                fix: Some("Report it at https://github.com/greshake/i3status-rust/issues.".into()),
1764            });
1765        }
1766        LiveVerdict::BlockError(error) => {
1767            // A block that failed to prepare already has a static problem
1768            // with the same root cause; do not diagnose it twice.
1769            if !prepared_errors.contains(label) {
1770                problems.push(Problem {
1771                    diagnosis: format!("{}: {}", report.name, escape_control(error)),
1772                    fix: suggest_fix(&report.name, error),
1773                });
1774            }
1775        }
1776        LiveVerdict::Panicked(msg) => {
1777            problems.push(Problem {
1778                diagnosis: format!(
1779                    "{}: panicked ({}). In the bar this would show a permanent error.",
1780                    report.name,
1781                    escape_control(msg)
1782                ),
1783                fix: Some(
1784                    "This is a bug in i3status-rs; please report it upstream with your block \
1785                     config."
1786                        .into(),
1787                ),
1788            });
1789        }
1790        LiveVerdict::IfCommandFailed(reason) => {
1791            problems.push(Problem {
1792                diagnosis: format!(
1793                    "{}: {} — the bar fails to start when if_command cannot run.",
1794                    report.name,
1795                    escape_control(reason)
1796                ),
1797                fix: Some("Fix or remove the if_command.".into()),
1798            });
1799        }
1800        _ => (),
1801    }
1802}

1804fn suggest_fix(block: &str, error: &str) -> Option<String> {
1805    let lower = error.to_lowercase();
1806    // D-Bus first: a missing bus socket also reads "no such file".
1807    if lower.contains("dbus") || lower.contains("d-bus") || lower.contains("org.freedesktop") {
1808        Some(format!(
1809            "A D-Bus service the `{block}` block needs is not available. Start the corresponding \
1810             daemon, or remove the block."
1811        ))
1812    } else if lower.contains("no such file") || lower.contains("program not found") {
1813        Some(format!(
1814            "A program the `{block}` block runs is not installed. Install it, or remove the block."
1815        ))
1816    } else if lower.contains("api key") || lower.contains("unauthorized") || lower.contains("401") {
1817        Some(format!(
1818            "Configure a valid API key for the `{block}` block."
1819        ))
1820    } else if lower.contains("rate limit") {
1821        Some("The service is rate limiting; wait, or reduce the update frequency.".into())
1822    } else {
1823        None
1824    }
1825}

1839/// Authoritative static analysis derived from a block's prepared contract
1840/// (see [`crate::block_plan`]). Runtime and doctor consume the same effective
1841/// formats and declared icon choices, so this cannot drift from what the
1842/// block actually does.
1843#[derive(Default)]
1844struct PlanAnalysis {
1845    /// Every icon name the block's contract declares, plus every direct
1846    /// `^icon_*` token in its effective formats. Blocks can only request
1847    /// declared icons (capability tokens), so all of these count as "in use
1848    /// by the block" — no reachability claims are made, and a declared icon
1849    /// the configuration never renders is harmless.
1850    required: HashSet<String>,
1851    /// Direct `^icon_*` names (subset of `required`).
1852    direct: HashSet<String>,
1853    /// The block declares an open dynamic icon source (custom/custom_dbus):
1854    /// any name may be requested at runtime, so static coverage is
1855    /// inherently partial for this block.
1856    open: bool,
1857}
1858
1859fn analyze_plan(plan: &crate::block_plan::BlockPlan) -> PlanAnalysis {
1860    use crate::block_plan::IconChoices;
1861    let mut analysis = PlanAnalysis::default();
1862    for output in plan.outputs() {
1863        for (_, choices) in output.icon_placeholders() {
1864            match choices {
1865                IconChoices::Fixed(names) => analysis.required.extend(
1866                    names
1867                        .iter()
1868                        .filter(|n| !n.is_empty())
1869                        .map(|n| n.to_string()),
1870                ),
1871                IconChoices::OpenResolvable => analysis.open = true,
1872            }
1873        }
1874        for template in [&output.format().full, &output.format().short] {
1875            let mut icons = Vec::new();
1876            collect_icon_tokens(template, &mut icons);
1877            for name in icons.into_iter().filter(|name| !name.is_empty()) {
1878                analysis.required.insert(name.clone());
1879                analysis.direct.insert(name);
1880            }
1881        }
1882    }
1883    analysis
1884}

1886/// Every direct `^icon_*` token in a compiled template, without branch
1887/// analysis: doctor makes no reachability claims.
1888fn collect_icon_tokens(template: &format_template::FormatTemplate, icon_refs: &mut Vec<String>) {
1889    for token_list in template.0.iter() {
1890        for token in &token_list.0 {
1891            match token {
1892                format_template::Token::Icon { name } => icon_refs.push(name.clone()),
1893                format_template::Token::Recursive(rec) => collect_icon_tokens(rec, icon_refs),
1894                format_template::Token::Placeholder { .. } | format_template::Token::Text(_) => (),
1895            }
1896        }
1897    }
1898}

1900struct IconTableInput<'a> {

1902    global_overrides: &'a HashMap<String, Icon>,
1903    block_overrides: &'a [(String, HashMap<String, Icon>)],
1904    used_now: &'a BTreeMap<String, HashSet<String>>,
1905    /// Icon name -> labels of block instances whose contract declares it.
1906    declared_by: &'a BTreeMap<String, Vec<String>>,
1907    /// (label, icon) pairs already diagnosed by a live render error.
1908    live_reported: &'a HashSet<(String, String)>,
1909    /// Per block label: its effective icons_format (the wrapper applied to
1910    /// every icon it renders).
1911    icons_formats: &'a HashMap<String, String>,
1912    /// Block labels whose effective icons_format contains pango markup:
1913    /// the font drawing their icons is not verified (inconclusive).
1914    markup_labels: &'a HashSet<String>,
1915    font_check: &'a mut Option<FontCheck>,
1916    font_authoritative: bool,
1917    style: &'a Style,
1918    problems: &'a mut Vec<Problem>,
1919    /// Set when there is nothing to list, e.g. the config does not validate.
1920    skip_reason: Option<&'static str>,
1921    display: IconDisplay,

1924fn print_icon_table(input: IconTableInput) {
1925    let IconTableInput {
1926        base_map,
1927        global_overrides,
1928        block_overrides,
1929        used_now,
1930        declared_by,
1931        live_reported,
1932        icons_formats,
1933        markup_labels,
1934        font_check,
1935        font_authoritative,
1936        style,
1937        problems,
1938        skip_reason,
1939        display,
1940    } = input;
1941    // Per-instance local overrides, keyed by instance label ("time#2"), so
1942    // two blocks of the same type are analyzed separately.
1943    let mut local: HashMap<&str, HashMap<&str, &Icon>> = HashMap::new();
1944    for (block, overrides) in block_overrides {
1945        let entry = local.entry(block.as_str()).or_default();
1946        for (name, icon) in overrides {
1947            entry.insert(name.as_str(), icon);
1948        }
1949    }
1950
1951    // What a given block instance actually resolves an icon name to.
1952    let resolve = |icon: &str, block: &str| -> Option<(&Icon, Provenance)> {
1953        if let Some(found) = local.get(block).and_then(|o| o.get(icon)) {
1954            return Some((found, Provenance::Local(block.to_string())));
1955        }
1956        if let Some(found) = global_overrides.get(icon) {
1957            return Some((found, Provenance::Global));
1958        }
1959        base_map.get(icon).map(|found| (found, Provenance::Base))
1960    };
1961
1962    // usage: icon name -> block labels. Declared = in use, period: every
1963    // icon a block's contract declares counts, whether or not this
1964    // particular run (or configuration) rendered it. Live-observed names
1965    // are merged in for the dynamic blocks.
1966    let mut usage: BTreeMap<String, Vec<String>> = BTreeMap::new();
1967    for (icon, blocks) in used_now {
1968        for block in blocks {
1969            usage.entry(icon.clone()).or_default().push(block.clone());
1970        }
1971    }
1972    for (icon, blocks) in declared_by {
1973        for block in blocks {
1974            let users = usage.entry((*icon).to_string()).or_default();
1975            if !users.contains(block) {
1976                users.push((*block).to_string());
1977            }
1978        }
1979    }
1980
1981    let mut used_rows: Vec<IconRow> = Vec::new();
1982    // Declared icon names that do not resolve for a block (not in the icon
1983    // set or any applicable override): listed as information.
1984    let mut undefined: Vec<(String, String)> = Vec::new();
1985    let mut fallback_rows = 0usize;
1986    let mut missing_rows = 0usize;
1987
1988    for (icon_name, users) in &usage {
1989        // Group the using blocks by what the icon actually resolves to for
1990        // them (an override REPLACES the base glyph for its block) and by
1991        // whether their icons_format markup makes the font inconclusive.
1992        let mut groups: BTreeMap<(Provenance, bool, String), Vec<String>> = BTreeMap::new();
1993        for block in users {
1994            match resolve(icon_name, block) {
1995                Some((icon, provenance)) => {
1996                    if matches!(icon, Icon::Progression(steps) if steps.is_empty()) {
1997                        // An empty progression cannot satisfy a request:
1998                        // Icons::get returns None for it, exactly like an
1999                        // undefined name.
2000                        if !live_reported.contains(&(block.clone(), icon_name.clone()))
2001                            && !undefined
2002                                .iter()
2003                                .any(|(name, b)| name == icon_name && b == block)
2004                        {
2005                            undefined.push((icon_name.clone(), block.clone()));
2006                        }
2007                        continue;
2008                    }
2009                    let markup = markup_labels.contains(block);
2010                    let icons_format = icons_formats
2011                        .get(block)
2012                        .cloned()
2013                        .unwrap_or_else(|| "{icon}".to_string());
2014                    groups
2015                        .entry((provenance, markup, icons_format))
2016                        .or_default()
2017                        .push(block.clone());
2018                }
2019                // already diagnosed by the live render error for this block
2020                None if live_reported.contains(&(block.clone(), icon_name.clone())) => (),
2021                None => {
2022                    // Availability information, not a problem: doctor makes
2023                    // no claim about whether the block's states and formats
2024                    // will actually request it — the live test reports
2025                    // failures that really happen.
2026                    if !undefined
2027                        .iter()
2028                        .any(|(name, b)| name == icon_name && b == block)
2029                    {
2030                        undefined.push((icon_name.clone(), block.clone()));
2031                    }
2032                }
2033            }
2034        }
2035        if groups.is_empty() {
2036            continue;
2037        }
2038        for ((provenance, markup, icons_format), mut blocks) in groups {
2039            blocks.sort_unstable();
2040            let (icon, tag) = match &provenance {
2041                Provenance::Base => (base_map.get(icon_name), String::new()),
2042                Provenance::Global => (global_overrides.get(icon_name), " [override]".into()),
2043                Provenance::Local(block) => (
2044                    local
2045                        .get(block.as_str())
2046                        .and_then(|o| o.get(icon_name.as_str()))
2047                        .copied(),
2048                    format!(" [{block} override]"),
2049                ),
2050            };
2051            let Some(icon) = icon else { continue };
2052            push_icon_rows(
2053                &mut used_rows,
2054                icon_name,
2055                icon,
2056                &tag,
2057                &blocks.join(", "),
2058                &icons_format,
2059                markup,
2060                font_check,
2061                &mut fallback_rows,
2062                &mut missing_rows,
2063            );
2064        }
2065    }
2066
2067    // Without a validated config there are no plans, so the table would be
2068    // its headers alone.
2069    if let Some(reason) = skip_reason {
2070        println!("Icons referenced by your blocks: skipped ({reason})");
2071    } else if used_rows.is_empty() {
2072        println!("Icons referenced by your blocks: none");
2073    } else {
2074        println!("Icons referenced by your blocks");
2075        let name_w = column_width("Name", used_rows.iter().map(|r| r.name.as_str()));
2076        let codes_w = column_width("Code", used_rows.iter().map(|r| r.codes.as_str()));
2077        let provider_w = column_width(
2078            "Effectively provided by",
2079            used_rows.iter().map(|r| r.provider.as_str()),
2080        );
2081        // The glyph cell renders as "X" including the quotes; size it by
2082        // display width so multi-character text icons ("LO") stay aligned.
2083        let glyph_w = used_rows
2084            .iter()
2085            .map(|r| UnicodeWidthStr::width(r.glyph.as_str()) + 2)
2086            .chain([UnicodeWidthStr::width("Glyph")])
2087            .max()
2088            .unwrap_or(5);
2089        // With names the glyph column would only repeat the Name column.
2090        let glyph_col = |glyph: &str| match display {
2091            IconDisplay::Glyphs => format!("{}  ", pad_right(glyph, glyph_w)),
2092            IconDisplay::Names => String::new(),
2093        };
2094        println!(
2095            "{}  {}{}  {}  Used by",
2096            pad_right("Name", name_w),
2097            glyph_col("Glyph"),
2098            pad_right("Code", codes_w),
2099            pad_right("Effectively provided by", provider_w)
2100        );
2101        for row in &used_rows {
2102            let line = format!(
2103                "{}  {}{}  {}  {}",
2104                pad_right(&row.name, name_w),
2105                glyph_col(&format!("\"{}\"", row.glyph)),
2106                pad_right(&row.codes, codes_w),
2107                pad_right(&row.provider, provider_w),
2108                row.used_by
2109            );
2110            if row.red {
2111                println!("{}{line}{}", style.red, style.reset);
2112            } else {
2113                println!("{line}");
2114            }
2115        }
2116    }
2117    if fallback_rows > 0 || missing_rows > 0 || !undefined.is_empty() || !markup_labels.is_empty() {
2118        println!();
2119    }
2120    if fallback_rows > 0 {
2121        println!("* Glyph provider selected by the system because no font in your list has it.");
2122    }
2123    if missing_rows > 0 {
2124        println!("† No installed font has this glyph; it renders as an empty box.");
2125    }
2126    if !undefined.is_empty() {
2127        let listed: Vec<String> = undefined
2128            .iter()
2129            .map(|(name, block)| format!("{} ({block})", escape_control(name)))
2130            .collect();
2131        // A block can only ever request icons its contract declares, so a
2132        // declared name with no definition is a real gap: whenever the
2133        // block reaches that state, the bar shows an error instead of the
2134        // block.
2135        problems.push(Problem {
2136            diagnosis: format!(
2137                "Icon(s) declared by a block but not defined in your icon set or overrides: {}.",
2138                listed.join(", ")
2139            ),
2140            fix: Some(
2141                "Add the name(s) to [icons.overrides] (or the block's icons_overrides), or use \
2142                 an icon set that defines them."
2143                    .into(),
2144            ),
2145        });
2146    }
2147    if !markup_labels.is_empty() {
2148        let mut listed: Vec<&str> = markup_labels.iter().map(String::as_str).collect();
2149        listed.sort_unstable();
2150        println!(
2151            "note: pango markup (from a format, icons_format, or an icon value) affects: {}.\n\
2152             The font drawing those blocks' icons depends on that markup and is not verified\n\
2153             by doctor.",
2154            listed.join(", ")
2155        );
2156    }
2157    println!();
2158
2159    if fallback_rows > 0 {
2160        let diagnosis = format!(
2161            "{fallback_rows} icon glyph(s) will be drawn by fonts outside the bar's font \
2162             list (red rows above). Their appearance depends on which fonts happen to be \
2163             installed and can change with any font install or system update."
2164        );
2165        if font_authoritative {
2166            problems.push(Problem {
2167                diagnosis,
2168                fix: Some(
2169                    "Add a font that provides these glyphs to the bar's font directive \
2170                     (matching the icon set in use), and make sure it is installed."
2171                        .into(),
2172                ),
2173            });
2174        } else {
2175            println!("note: {diagnosis}");
2176            println!("note: the bar font was guessed, so this is not counted as a problem.");
2177            println!();
2178        }
2179    }
2180    if missing_rows > 0 {
2181        problems.push(Problem {
2182            diagnosis: format!(
2183                "{missing_rows} icon glyph(s) have no provider at all († rows above)."
2184            ),
2185            fix: Some(
2186                "Install a font that contains these codepoints, or override the icons with \
2187                 glyphs your fonts have."
2188                    .into(),
2189            ),
2190        });
2191    }
2192}

2194/// Expand an icon into one table row per glyph (progressions as "name k/n")
2195/// and classify each glyph's font provider.
2196#[allow(clippy::too_many_arguments)]
2197fn push_icon_rows(
2198    used_rows: &mut Vec<IconRow>,
2199    name: &str,
2200    icon: &Icon,
2201    tag: &str,
2202    used_by: &str,
2203    icons_format: &str,
2204    markup: bool,
2205    font_check: &mut Option<FontCheck>,
2206    fallback_rows: &mut usize,
2207    missing_rows: &mut usize,
2208) {
2209    let name = escape_control(name);
2210    let rows: Vec<(String, &str)> = match icon {
2211        Icon::Single(s) => vec![(name.clone(), s.as_str())],
2212        Icon::Progression(steps) => steps
2213            .iter()
2214            .enumerate()
2215            .map(|(k, s)| (format!("{name} {}/{}", k + 1, steps.len()), s.as_str()))
2216            .collect(),
2217    };
2218    for (row_name, glyph) in rows {
2219        // What the bar actually draws: icons_format wraps every icon, and
2220        // whatever it adds is rendered with the same font questions.
2221        let effective = icons_format.replace("{icon}", glyph);
2222        let codes = codepoints(&effective);
2223        // An icon value containing markup is not a plain glyph: fontconfig
2224        // cannot be asked which font draws it.
2225        let markup = markup || contains_markup(&effective);
2226        let (provider, mark) = if markup {
2227            // Pango markup applies to this icon; doctor does not emulate
2228            // pango, so which font draws the glyph is unknown.
2229            ("(depends on pango markup)".to_string(), "")
2230        } else {
2231            match font_check.as_mut() {
2232                None => ("?".to_string(), ""),
2233                Some(check) => match glyph_provider(check, &effective) {
2234                    GlyphProvider::Known(family) => (escape_control(&first_family(&family)), ""),
2235                    GlyphProvider::Fallback(family) => {
2236                        *fallback_rows += 1;
2237                        (escape_control(&first_family(&family)), " *")
2238                    }
2239                    GlyphProvider::Missing => {
2240                        *missing_rows += 1;
2241                        ("(none)".to_string(), " †")
2242                    }
2243                },
2244            }
2245        };
2246        used_rows.push(IconRow {
2247            name: row_name,
2248            // Icon values are user-supplied: never let one emit terminal
2249            // control sequences into the table.
2250            glyph: escape_control(&effective),
2251            codes,
2252            provider: format!("{provider}{mark}{tag}"),
2253            used_by: used_by.to_string(),
2254            red: !mark.is_empty(),
2255        });
2256    }
2257}

2268/// Pad `text` with spaces until it occupies `width` terminal cells. The
2269/// standard formatter pads by character count, which misaligns any table
2270/// containing wide (CJK, emoji) or zero-width characters.
2271fn pad_right(text: &str, width: usize) -> String {
2272    let padding = width.saturating_sub(UnicodeWidthStr::width(text));
2273    format!("{text}{}", " ".repeat(padding))

2277    values
2278        .map(UnicodeWidthStr::width)
2279        .chain([UnicodeWidthStr::width(header)])
2280        .max()
2281        .unwrap_or(0)

2284fn codepoints(s: &str) -> String {
2285    let codes: Vec<String> = s
2286        .chars()
2287        .filter(|c| !c.is_ascii())
2288        .map(|c| format!("U+{:04X}", c as u32))
2289        .collect();
2290    if codes.is_empty() {
2291        "-".to_string()
2292    } else {
2293        codes.join(" ")
2294    }
2295}

2304/// fc-match reports family lists like "Font Awesome 5 Free,Font Awesome 5
2305/// Free Solid"; the first member is enough for display.
2306fn first_family(family: &str) -> String {
2307    family.split(',').next().unwrap_or(family).to_string()

2310/// Unicode's `Default_Ignorable_Code_Point` property: characters a
2311/// renderer is expected to draw nothing for — joiners, variation
2312/// selectors, bidi controls, fillers, tags. Answered by ICU's Unicode
2313/// data rather than by a table maintained here.
2314///
2315/// This is not the same question as the general category: U+115F HANGUL
2316/// CHOSEONG FILLER is a letter that draws nothing, while a combining
2317/// acute accent has no advance width but a very visible glyph.
2318fn is_default_ignorable(c: char) -> bool {
2319    CodePointSetData::new::<DefaultIgnorableCodePoint>().contains(c)
2320}
2321
2322/// Interlinear annotation controls: `Cf` characters that are not
2323/// default-ignorable but are still an editing protocol rather than text.
2324fn is_annotation_char(c: char) -> bool {
2325    matches!(c, '\u{fff9}'..='\u{fffb}')
2326}

2328/// Characters with no glyph of their own, which fontconfig cannot
2329/// meaningfully be asked about.
2330fn is_invisible(c: char) -> bool {
2331    is_default_ignorable(c) || is_annotation_char(c)
2332}

2334/// Whether a character has a glyph a font must supply. Zero *advance
2335/// width* is not the test — combining marks draw perfectly visible
2336/// accents — so this asks Unicode which characters are ignorable.
2337fn is_drawn(c: char) -> bool {
2338    !c.is_control() && !c.is_whitespace() && !is_invisible(c)
2339}

2341fn glyph_provider(check: &mut FontCheck, glyph: &str) -> GlyphProvider {
2342    let mut result = GlyphProvider::Known(check.base_family.clone());
2343    for c in glyph.chars() {
2344        if !is_drawn(c) {
2345            continue;
2346        }
2347        match check.check(c) {
2348            GlyphFont::Base => (),
2349            GlyphFont::Configured(family) => {
2350                if matches!(result, GlyphProvider::Known(_)) {
2351                    result = GlyphProvider::Known(family.clone());
2352                }
2353            }
2354            GlyphFont::Fallback(family) => {
2355                if !matches!(result, GlyphProvider::Missing) {
2356                    result = GlyphProvider::Fallback(family.clone());
2357                }
2358            }
2359            GlyphFont::Missing => result = GlyphProvider::Missing,
2360        }
2361    }
2362    result
2363}

2392/// Returns the winning path and a human-readable trace of every candidate.
2393/// What the bar's own lookup would do with this name.
2394enum Resolved {
2395    /// The file the bar would use.
2396    Found(PathBuf),
2397    /// No candidate exists.
2398    Missing,
2399    /// A candidate could not be examined. [`util::find_file`] returns the
2400    /// error rather than moving on, so the bar never reaches the
2401    /// lower-priority candidates and neither may doctor.
2402    Failed(String),
2403}
2404
2405fn resolve_file(file: &str, subdir: Option<&str>) -> (Resolved, String) {
2406    let mut found: Option<PathBuf> = None;
2407    let mut failure: Option<String> = None;
2408    let mut trace = String::new();
2409    if !Path::new(file).is_absolute() {
2410        trace.push_str("   (not an absolute path; standard locations searched in order)\n");
2411    }
2412    for candidate in util::file_candidates(file, subdir, Some("toml")) {
2413        let line = match (candidate.try_exists(), &found) {
2414            (Ok(true), None) => {
2415                let line = format!(
2416                    "   ✓ {} ← using this\n",
2417                    escape_control(&candidate.display().to_string())
2418                );
2419                found = Some(candidate);
2420                line
2421            }
2422            // Candidates embed the configured name: a newline in it must
2423            // not forge a line in doctor's own report.
2424            (Ok(true), Some(_)) => format!(
2425                "   ✓ {} (shadowed)\n",
2426                escape_control(&candidate.display().to_string())
2427            ),
2428            (Ok(false), _) => format!(
2429                "   ✗ {}\n",
2430                escape_control(&candidate.display().to_string())
2431            ),
2432            (Err(err), _) => {
2433                let line = format!(
2434                    "   ✗ {} (cannot be examined: {err} — the bar stops here)\n",
2435                    escape_control(&candidate.display().to_string())
2436                );
2437                failure = Some(format!("{}: {err}", candidate.display()));
2438                line
2439            }
2440        };
2441        trace.push_str(&line);
2442        if failure.is_some() {
2443            break;
2444        }
2445    }
2446    let resolved = match (failure, found) {
2447        (Some(err), _) => Resolved::Failed(err),
2448        (None, Some(path)) => Resolved::Found(path),
2449        (None, None) => Resolved::Missing,
2450    };
2451    (resolved, trace.trim_end().to_string())
2452}

2471/// Asks fontconfig (via `fc-match`/`fc-list`) which font actually renders
2472/// each glyph. This happens in the bar, outside of i3status-rs, and is the
2473/// usual source of "my icon was silently replaced by a different symbol":
2474/// when no configured font has a codepoint, fontconfig substitutes another
2475/// installed font without telling anyone.
2476struct FontCheck {
2477    /// The family that draws plain text: the first configured family that is
2478    /// actually present.
2479    base_family: String,
2480    /// Index of that family in `families`.
2481    base_index: Option<usize>,
2482    /// Each bar family with what fc-match resolves it to; a family that
2483    /// resolves to something else entirely is not installed.
2484    families: Vec<(String, Option<String>)>,
2485    cache: HashMap<char, GlyphFont>,
2486}
2487
2488impl FontCheck {
2489    fn new(font_arg: Option<&str>) -> Option<Self> {
2490        let families = parse_font_directive(font_arg.unwrap_or(""));
2491        let pattern = families.join(",");
2492        // Also proves fc-match exists; used as the base family only when the
2493        // bar configured no families of its own.
2494        let fontconfig_default = fc_match(&pattern, None)?;
2495        // fontconfig ships fc-list with fc-match; without both, no check.
2496        let has_fc_list = std::process::Command::new("fc-list")
2497            .arg("--version")
2498            .output()
2499            .is_ok_and(|out| out.status.success());
2500        if !has_fc_list {
2501            return None;
2502        }
2503        let mut families: Vec<(String, Option<String>)> = families
2504            .into_iter()
2505            .map(|family| {
2506                let resolved = fc_match(&family, None);
2507                (family, resolved)
2508            })
2509            .collect();
2510        if families.is_empty() {
2511            // No font directive: the bar renders with fontconfig's default,
2512            // so that is the one family it "configured".
2513            families.push((fontconfig_default.clone(), Some(fontconfig_default.clone())));
2514        }
2515        let base_index = families
2516            .iter()
2517            .position(|(family, _)| family_present(family));
2518        let base_family = match base_index {
2519            Some(index) => {
2520                let (family, resolved) = &families[index];
2521                // A generic alias is not a family any font carries; report
2522                // the concrete family it stands for.
2523                if is_generic_family(family) {
2524                    resolved
2525                        .as_deref()
2526                        .map_or_else(|| family.clone(), first_family)
2527                } else {
2528                    family.clone()
2529                }
2530            }
2531            None => fontconfig_default,
2532        };
2533        Some(Self {
2534            base_family,
2535            base_index,
2536            families,
2537            cache: HashMap::new(),
2538        })
2539    }
2540
2541    fn check(&mut self, c: char) -> GlyphFont {
2542        if let Some(hit) = self.cache.get(&c) {
2543            return hit.clone();
2544        }
2545        let result = self.classify(c);
2546        self.cache.insert(c, result.clone());
2547        result
2548    }
2549
2550    fn classify(&self, c: char) -> GlyphFont {
2551        let charset = format!("{:x}", c as u32);
2552        if !fc_list_provides(&charset) {
2553            return GlyphFont::Missing;
2554        }
2555        // The bar falls back per character down its font list, so the first
2556        // configured family that has the codepoint is the one that draws it.
2557        // Asking fontconfig to match the whole list at once answers a
2558        // different question — it ranks by its own rules, and a color emoji
2559        // font in the list outranks family order for every codepoint, which
2560        // made every glyph look like it came from that one font.
2561        for (index, (family, resolved)) in self.families.iter().enumerate() {
2562            if !family_covers(family, resolved, &charset) {
2563                continue;
2564            }
2565            return if Some(index) == self.base_index {
2566                GlyphFont::Base
2567            } else {
2568                GlyphFont::Configured(family.clone())
2569            };
2570        }
2571        // No configured family has it: whatever fontconfig substitutes is
2572        // what the user will actually see, and is not what they asked for.
2573        match fc_match("", Some(&charset)) {
2574            Some(family) => GlyphFont::Fallback(family),
2575            None => GlyphFont::Missing,
2576        }
2577    }
2578}

2580/// Whether one configured family has this codepoint, asked of fontconfig:
2581/// `fc-list` with a family and a charset lists only fonts that satisfy both.
2582///
2583/// A generic alias (`monospace`, ...) is not a family any font carries, so it
2584/// is tested through the concrete family it resolves to. A family that is not
2585/// installed also resolves to something, but that substitute is not what the
2586/// user configured, so it is not consulted.
2587fn family_covers(family: &str, resolved: &Option<String>, charset: &str) -> bool {
2588    if fc_list_family_provides(family, charset) {
2589        return true;
2590    }
2591    is_generic_family(family)
2592        && resolved.as_ref().is_some_and(|resolved| {
2593            resolved
2594                .split(',')
2595                .any(|concrete| fc_list_family_provides(concrete, charset))
2596        })
2597}

2599/// Whether a configured family is present at all. Generic aliases always
2600/// resolve to something.
2601fn family_present(family: &str) -> bool {
2602    is_generic_family(family) || family_installed(family)
2603}

2605/// Whether a family is installed, asked of fontconfig itself: `fc-list` with
2606/// a family pattern lists only fonts whose family matches under fontconfig's

2611        .arg("family")
2612        .output()
2613        .is_ok_and(|out| {
2614            out.status.success() && !String::from_utf8_lossy(&out.stdout).trim().is_empty()
2615        })
2616}

2636fn fc_match(pattern: &str, charset: Option<&str>) -> Option<String> {
2637    let mut pattern = pattern.to_string();

2639        pattern.push_str(&format!(":charset={charset}"));
2640    }

2647        return None;
2648    }
2649    let family = String::from_utf8_lossy(&out.stdout).trim().to_string();
2650    (!family.is_empty()).then_some(family)
2651}

2679/// Ask the running i3/sway over IPC which font the bar is configured with,
2680/// so `--font` is only needed when there is no live bar to ask.
2681///
2682/// With several bars, prefer the one whose status_command runs i3status-rs;
2683/// if the remaining candidates disagree on the font, report the ambiguity.
2684fn detect_bar_font() -> Option<DetectedFont> {

2686    // i3 or sway through I3SOCK, SWAYSOCK or `--get-socketpath`.
2687    let mut ipc = swayipc::Connection::new().ok()?;
2688
2689    // (bar_id, font, status_command)
2690    let bars: Vec<(String, String, String)> = ipc
2691        .get_bar_ids()
2692        .ok()?
2693        .into_iter()
2694        .filter_map(|id| ipc.get_bar_config(&id).ok())
2695        .map(|bar| (bar.id, bar.font, bar.status_command.unwrap_or_default()))
2696        .collect();

2698        return None;
2699    }

2702    let (bar_id, font, _) = bars.into_iter().nth(chosen)?;

2704}

2724/// The bar whose font to check, as `(index, note)` into `bars`, which holds
2725/// `(bar_id, font, status_command)`. Only bars running i3status-rs are
2726/// candidates; with none, every bar is. The note warns when the candidates
2727/// disagree on the font.
2728fn pick_bar(bars: &[(String, String, String)]) -> (usize, Option<String>) {
2729    let ours: Vec<usize> = (0..bars.len())
2730        .filter(|&i| bars[i].2.contains("i3status-rs"))
2731        .collect();
2732    let candidates = if ours.is_empty() {
2733        (0..bars.len()).collect()
2734    } else {
2735        ours
2736    };
2737
2738    let chosen = candidates[0];
2739    let font = &bars[chosen].1;
2740    if candidates.iter().all(|&i| bars[i].1 == *font) {
2741        return (chosen, None);
2742    }
2743
2744    let list = candidates
2745        .iter()
2746        .map(|&i| format!("{}: {:?}", bars[i].0, bars[i].1))
2747        .collect::<Vec<_>>()
2748        .join(", ");
2749    let note = format!(
2750        "several bars with different fonts ({list}); picked {:?} — pass --font if that \
2751         is the wrong one",
2752        bars[chosen].0
2753    );
2754    (chosen, Some(note))
2755}

2757/// The font families an i3/sway font directive names, in order.
2758///
2759/// `"pango:DejaVu Sans Mono, Font Awesome 6 Free Bold 12"` →
2760/// `["DejaVu Sans Mono", "Font Awesome 6 Free"]`
2761///
2762/// (the style and size qualify the whole description, so they follow the
2763/// fallback list rather than the first family)
2764///
2765/// This reads the description the way `pango_font_description_from_string`
2766/// does, from the end: optional `@variations`, then a size (`12`, `16px`),
2767/// then style words; what is left is the comma-separated family list. A test
2768/// checks it against pango itself.
2769fn parse_font_directive(raw: &str) -> Vec<String> {
2770    let mut rest = raw.strip_prefix("pango:").unwrap_or(raw);
2771
2772    // Variations start a word and may hold commas (`@wght=600,wdth=80`), so
2773    // only whitespace ends them.
2774    let trimmed = rest.trim_end();
2775    let word_start = trimmed.rfind(char::is_whitespace).map_or(0, |i| i + 1);
2776    if trimmed[word_start..].starts_with('@') {
2777        rest = &trimmed[..word_start];
2778    }
2779    if let Some((head, word)) = last_word(rest)
2780        && is_font_size(word)
2781    {
2782        rest = head;
2783    }
2784    while let Some((head, word)) = last_word(rest)
2785        && is_style_word(word)
2786    {
2787        rest = head;
2788    }
2789
2790    rest.split(',')
2791        .map(str::trim)
2792        .filter(|family| !family.is_empty())
2793        .map(String::from)
2794        .collect()
2795}

2797/// The last word of a font description and what precedes it. Words end at
2798/// whitespace or a comma, as in pango: `"Sans, Bold"` → `("Sans, ", "Bold")`.
2799fn last_word(description: &str) -> Option<(&str, &str)> {
2800    let trimmed = description.trim_end();
2801    let start = trimmed
2802        .rfind(|c: char| c.is_whitespace() || c == ',')
2803        .map_or(0, |i| i + 1);
2804    let word = &trimmed[start..];
2805    (!word.is_empty()).then(|| (&trimmed[..start], word))
2806}
