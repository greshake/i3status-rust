1//! Block output contracts.
2//!
3//! Every standard block prepares a [`BlockPlan`] from its configuration before
4//! running. The plan declares each output variant the block can produce, the
5//! effective format for that output (defaults and inheritance already
6//! resolved), and the icon names each icon-valued placeholder can carry. A
7//! block that renders plain runtime text rather than a format declares that
8//! too, as an [`OutputKind::Text`] output.
9//!
10//! The runtime renders through [`OutputHandle`]s derived from the plan, which
11//! install the effective format and can look up the icon names the plan
12//! declares. Icons are checked in one place, when a widget is published (see
13//! [`Widget::check_contract`][crate::widget::Widget]): every icon value it
14//! carries must be one the output declared, however that value was built.
15//!
16//! The invariant is that everything a block puts on the bar comes from a
17//! declared output: publishing a format or text widget without a handle of
18//! the matching kind fails, a plan that declares nothing is rejected when it
19//! is built, and [`Widget::set_format`][crate::widget::Widget] and
20//! [`Widget::set_text`][crate::widget::Widget] void the contract rather than
21//! quietly keeping it. A widget with no source at all renders nothing and is
22//! exempt.
23//!
24//! Because the plan is prepared before the block runs, a block's icon and
25//! format surface can also be inspected without running it, and what such an
26//! inspection sees cannot drift from what the block actually renders.
27//!
28//! Declaring an icon does not resolve it. Resolution stays lazy: an icon is
29//! looked up in the icon set only when a reachable format actually renders it,
30//! so configurations that omit icons their formats never reference keep
31//! working.
32

41/// Marks an error caused by a block breaking its own contract rather than
42/// by the user's configuration. Every contract failure is prefixed with it,
43/// so the two kinds of error can be told apart by their message.
44pub(crate) const CONTRACT_BUG: &str = "block contract bug";
45
46/// The icon names one placeholder of one output variant can carry.
47#[derive(Debug, Clone)]
48pub(crate) enum IconChoices {
49    /// A closed set, fully known once the block's configuration is parsed.
50    /// Includes configuration-derived names (e.g. `toggle`'s `icon_on`).
51    Fixed(Vec<Cow<'static, str>>),
52    /// Any icon name is permitted and resolves through the normal icon set
53    /// and override rules at render time. Reserved for deliberately dynamic
54    /// blocks (`custom`, `custom_dbus`); standard blocks declare closed sets.
55    OpenResolvable,
56}
57
58impl IconChoices {
59    pub(crate) fn fixed<I, S>(names: I) -> Self
60    where
61        I: IntoIterator<Item = S>,
62        S: Into<Cow<'static, str>>,
63    {
64        Self::Fixed(names.into_iter().map(Into::into).collect())
65    }
66
67    pub(crate) fn one<S: Into<Cow<'static, str>>>(name: S) -> Self {
68        Self::Fixed(vec![name.into()])
69    }
70
71    pub(crate) fn permits(&self, name: &str) -> bool {
72        // An empty icon name renders as empty output (a runtime no-op), so
73        // it is always permitted.
74        name.is_empty()
75            || match self {
76                Self::Fixed(names) => names.iter().any(|n| n == name),
77                Self::OpenResolvable => true,
78            }
79    }
80}

82/// What an output renders.
83#[derive(Debug, Clone, Copy, PartialEq, Eq)]
84pub(crate) enum OutputKind {
85    /// A format template: placeholders, and icons among them.
86    Format,
87    /// Plain text computed at runtime, with no placeholders and no icons
88    /// (`menu`). The text itself cannot be known before the block runs, but
89    /// declaring the output keeps the block inside the contract: it states
90    /// that this is all the block renders.
91    Text,
92}

94/// One output variant of a block: a named state, its effective format, and
95/// the icons each icon-valued placeholder may carry in that state.
96#[derive(Debug, Clone)]
97pub(crate) struct OutputPlan {
98    id: Cow<'static, str>,
99    kind: OutputKind,
100    format: Format,
101    icons: Vec<(&'static str, IconChoices)>,
102}
103
104impl OutputPlan {
105    pub(crate) fn new(id: impl Into<Cow<'static, str>>, format: Format) -> Self {
106        Self {
107            id: id.into(),
108            kind: OutputKind::Format,
109            format,
110            icons: Vec::new(),
111        }
112    }
113
114    /// An output that renders plain runtime text instead of a format.
115    pub(crate) fn text(id: impl Into<Cow<'static, str>>) -> Self {
116        Self {
117            id: id.into(),
118            kind: OutputKind::Text,
119            format: Format::new(
120                "".parse().expect("the empty template always parses"),
121                "".parse().expect("the empty template always parses"),
122            ),
123            icons: Vec::new(),
124        }
125    }
126
127    pub(crate) fn id(&self) -> &str {
128        &self.id
129    }
130
131    pub(crate) fn kind(&self) -> OutputKind {
132        self.kind
133    }
134
135    /// The effective format of this output. A [`OutputKind::Text`] output
136    /// renders no template, so its format is empty.
137    pub(crate) fn format(&self) -> &Format {
138        &self.format
139    }
140
141    /// Declare the icon choices for an icon-valued placeholder.
142    /// Duplicate declarations are rejected when the plan is built.
143    pub(crate) fn icon(mut self, placeholder: &'static str, choices: IconChoices) -> Self {
144        self.icons.push((placeholder, choices));
145        self
146    }
147
148    pub(crate) fn choices_for(&self, placeholder: &str) -> Option<&IconChoices> {
149        self.icons
150            .iter()
151            .find(|(p, _)| *p == placeholder)
152            .map(|(_, c)| c)
153    }
154
155    /// The icon-valued placeholders this output declares. The bar itself
156    /// renders through [`OutputHandle`] and looks placeholders up by name, so
157    /// nothing in the hot path needs this — it exists because a plan that can
158    /// only be queried by a name you already know is not inspectable. The
159    /// declaration tests are its consumer today.
160    #[cfg(test)]
161    pub(crate) fn icon_placeholders(&self) -> impl Iterator<Item = (&'static str, &IconChoices)> {
162        self.icons.iter().map(|(p, c)| (*p, c))
163    }
164
165    /// Every static `^icon_name` this output's format can render: both the
166    /// full and short templates, and every alternative branch within them.
167    ///
168    /// These are the other half of the icon surface. `net` and `speedtest`
169    /// spell their arrows and their ping icon straight into the format
170    /// rather than passing icon values, so they declare no icon placeholder
171    /// for them, and an inspector that only walked [`Self::icon_placeholders`]
172    /// would conclude the block draws no icons at all.
173    #[cfg(test)]
174    pub(crate) fn static_icons(&self) -> Vec<&str> {
175        use crate::formatting::template::{FormatTemplate, Token};
176
177        fn walk<'a>(template: &'a FormatTemplate, found: &mut Vec<&'a str>) {
178            for list in template.0.iter() {
179                for token in &list.0 {
180                    match token {
181                        Token::Icon { name } => {
182                            if !found.contains(&name.as_str()) {
183                                found.push(name);
184                            }
185                        }
186                        Token::Recursive(inner) => walk(inner, found),
187                        Token::Text(_) | Token::Placeholder { .. } => (),
188                    }
189                }
190            }
191        }
192        let mut found = Vec::new();
193        walk(&self.format.full, &mut found);
194        walk(&self.format.short, &mut found);
195        found
196    }
197}

199/// One [`OutputPlan`] per format the user configured, so a block that can
200/// rotate through several formats declares each of them. `declare` adds the
201/// icon choices, which are the same whichever format is on screen.
202///
203/// Use this when the block's config has a `MaybeMultiFormatConfig`: the
204/// number of outputs is not known until the config is parsed. A block with a
205/// single `format` has one output and builds it inline instead (see
206/// `docker`). This returns a `Vec` rather than a finished plan because a
207/// block can append outputs that are not format variants (see `net`).
208pub(crate) fn format_outputs(
209    formats: Vec<Format>,
210    declare: impl Fn(OutputPlan) -> OutputPlan,
211) -> Vec<OutputPlan> {
212    formats
213        .into_iter()
214        .enumerate()
215        .map(|(index, format)| declare(OutputPlan::new(format_id(index), format)))
216        .collect()
217}
218
219/// The output id of the `index`th configured format.
220pub(crate) fn format_id(index: usize) -> Cow<'static, str> {
221    match index {
222        0 => "format".into(),
223        n => format!("format{}", n + 1).into(),
224    }
225}

227/// Rotating over the outputs [`format_outputs`] produced, mirroring the
228/// bar's own next/prev format actions.
229pub(crate) struct FormatRotation {
230    outputs: Vec<OutputHandle>,
231    index: usize,
232}
233
234impl FormatRotation {
235    /// Collects the plan's format outputs. A block may declare further
236    /// outputs for states that are not format variants (net's `inactive`
237    /// and `missing`); those are left alone.
238    pub(crate) fn new(plan: &Arc<BlockPlan>) -> Result<Self> {
239        let outputs: Vec<OutputHandle> = (0..)
240            .map_while(|index| plan.output(&format_id(index)).ok())
241            .collect();
242        if outputs.is_empty() {
243            return Err(Error::new(format!(
244                "{CONTRACT_BUG}: no format outputs to rotate over"
245            )));
246        }
247        Ok(Self { outputs, index: 0 })
248    }
249
250    pub(crate) fn current(&self) -> &OutputHandle {
251        &self.outputs[self.index]
252    }
253
254    pub(crate) fn next(&mut self) {
255        self.index = (self.index + 1) % self.outputs.len();
256    }
257
258    pub(crate) fn prev(&mut self) {
259        self.index = (self.index + self.outputs.len() - 1) % self.outputs.len();
260    }
261}

263/// The full prepared contract of one block instance. The validated fields
264/// are immutable from outside this module (read-only iteration only), so
265/// construction-time uniqueness is a permanent invariant.
266/// Deliberately not `Default`: an empty plan is exactly what [`Self::new`]
267/// rejects, and a derived `Default` would hand every block a way around it.
268#[derive(Debug, Clone)]
269pub(crate) struct BlockPlan {
270    outputs: Vec<OutputPlan>,
271}
272
273impl BlockPlan {
274    /// Read-only view of the declared output variants.
275    pub(crate) fn outputs(&self) -> impl Iterator<Item = &OutputPlan> {
276        self.outputs.iter()
277    }
278
279    /// Build a plan, rejecting ambiguous or incomplete metadata
280    /// unconditionally (also in release builds): a plan that declares nothing,
281    /// duplicate output ids, duplicate icon-placeholder declarations within an
282    /// output, and icons on a text output are all contract bugs.
283    pub(crate) fn new(outputs: Vec<OutputPlan>) -> Result<Arc<Self>> {
284        // A block that declares no outputs would render whatever it liked
285        // with nothing to check it against, which is the one hole the
286        // contract exists to close.
287        if outputs.is_empty() {
288            return Err(Error::new(format!(
289                "{CONTRACT_BUG}: the plan declares no outputs"
290            )));
291        }
292        for (i, output) in outputs.iter().enumerate() {
293            if output.kind == OutputKind::Text && !output.icons.is_empty() {
294                return Err(Error::new(format!(
295                    "{CONTRACT_BUG}: text output '{}' declares icons, which it \
296                     cannot render",
297                    output.id
298                )));
299            }
300            if outputs[..i].iter().any(|o| o.id == output.id) {
301                return Err(Error::new(format!(
302                    "{CONTRACT_BUG}: duplicate output id '{}'",
303                    output.id
304                )));
305            }
306            for (j, (placeholder, _)) in output.icons.iter().enumerate() {
307                if output.icons[..j].iter().any(|(p, _)| p == placeholder) {
308                    return Err(Error::new(format!(
309                        "{CONTRACT_BUG}: output '{}' declares '${placeholder}' twice",
310                        output.id
311                    )));
312                }
313            }
314        }
315        Ok(Arc::new(Self { outputs }))
316    }
317
318    /// Handle for the output variant named `id`.
319    pub(crate) fn output(self: &Arc<Self>, id: &str) -> Result<OutputHandle> {
320        let index = self
321            .outputs
322            .iter()
323            .position(|o| o.id == id)
324            .or_error(|| format!("{CONTRACT_BUG}: the plan has no output variant '{id}'"))?;
325        Ok(OutputHandle {
326            plan: self.clone(),
327            index,
328        })
329    }
330}

332/// A handle to one declared output variant. Constructing a widget through the
333/// handle installs the plan's effective format, so runtime code cannot select
334/// a format the plan does not know about.
335#[derive(Debug, Clone)]
336pub(crate) struct OutputHandle {
337    plan: Arc<BlockPlan>,
338    index: usize,
339}
340
341impl OutputHandle {
342    pub(crate) fn output(&self) -> &OutputPlan {
343        &self.plan.outputs[self.index]
344    }
345
346    pub(crate) fn id(&self) -> &str {
347        self.output().id()
348    }
349
350    pub(crate) fn format(&self) -> &Format {
351        self.output().format()
352    }
353
354    /// The one icon name this output declares for `placeholder`. Lets runtime
355    /// code take the name from the plan instead of repeating the string.
356    pub(crate) fn single_icon(&self, placeholder: &str) -> Result<Cow<'static, str>> {
357        match self.output().choices_for(placeholder) {
358            Some(IconChoices::Fixed(names)) if names.len() == 1 => Ok(names[0].clone()),
359            _ => Err(Error::new(format!(
360                "{CONTRACT_BUG}: output '{}' does not declare exactly one icon \
361                 for '${placeholder}'",
362                self.id()
363            ))),
364        }
365    }
366
367    /// The single declared icon of `placeholder`, as a value.
368    pub(crate) fn icon_value(&self, placeholder: &str) -> Result<Value> {
369        let name = self.single_icon(placeholder)?;
370        Ok(Value::icon(name))
371    }
372
373    /// A widget rendering this output: effective format installed, icon
374    /// values checked against the declared choices. Publishing it fails if
375    /// this output is a [`OutputKind::Text`] one.
376    pub(crate) fn new_widget(&self) -> Widget {
377        Widget::from_output(self)
378    }
379
380    /// A widget rendering `text` as this output. Publishing it fails unless
381    /// the output was declared as [`OutputKind::Text`].
382    pub(crate) fn new_text_widget(&self, text: String) -> Widget {
383        Widget::text_from_output(self, text)
384    }
385
386    /// Descriptions of every icon value in `values` that this output does not
387    /// declare. Empty when the widget conforms to the contract.
388    pub(crate) fn icon_violations(&self, values: &crate::formatting::Values) -> Vec<String> {
389        let output = self.output();
390        let mut violations = Vec::new();
391        for (key, value) in values {
392            if let ValueInner::Icon(name, _) = &value.inner {
393                let ok = output
394                    .choices_for(key)
395                    .is_some_and(|choices| choices.permits(name));
396                if !ok {
397                    violations.push(format!(
398                        "{CONTRACT_BUG}: icon '{name}' set for placeholder '${key}' \
399                         is not declared by output '{}'",
400                        output.id
401                    ));
402                }
403            }
404        }
405        violations
406    }
407}

409/// The error outputs every block shares: the effective (global or per-block)
410/// error formats with the standard error placeholders, including the
411/// conditional `refresh` icon shown for restartable errors. The bar renders
412/// every block error through these handles.
413#[derive(Debug)]
414pub(crate) struct ErrorOutputs {
415    pub error: OutputHandle,
416    pub fullscreen: OutputHandle,
417}
418
419pub(crate) fn error_outputs(
420    error_format: Format,
421    error_fullscreen_format: Format,
422    restartable_possible: bool,
423) -> ErrorOutputs {
424    let plan = error_plan(error_format, error_fullscreen_format, restartable_possible)
425        .expect("the error plan has statically unique ids");
426    ErrorOutputs {
427        error: OutputHandle {
428            plan: plan.clone(),
429            index: 0,
430        },
431        fullscreen: OutputHandle { plan, index: 1 },
432    }
433}
434
435/// The plan behind [`error_outputs`]: output 0 is "error", output 1 is
436/// "error_fullscreen". The conditional `refresh` restart icon is only
437/// declared when the block can actually become restartable: without a
438/// `max_retries` limit the bar retries forever and never renders the
439/// restart button.
440pub(crate) fn error_plan(
441    error_format: Format,
442    error_fullscreen_format: Format,
443    restartable_possible: bool,
444) -> Result<Arc<BlockPlan>> {
445    let output = |id, format| {
446        let output = OutputPlan::new(id, format);
447        if restartable_possible {
448            output.icon(
449                "restart_block_icon",
450                IconChoices::one(crate::icons::REFRESH),
451            )
452        } else {
453            output
454        }
455    };
456    BlockPlan::new(vec![
457        output("error", error_format),
458        output("error_fullscreen", error_fullscreen_format),
459    ])
460}

611    #[test]
612    fn a_plan_that_declares_nothing_is_rejected() {
613        // `BlockPlan::new` is the only way to build a plan. A `Default` impl
614        // (derived or written) would be a second, unchecked constructor
615        // producing exactly the empty plan this rejects, so assert at compile
616        // time that there is none: the two implementations below overlap if,
617        // and only if, `BlockPlan: Default`.
618        trait OnlyBuiltByNew {}
619        impl<T: Default> OnlyBuiltByNew for T {}
620        impl OnlyBuiltByNew for BlockPlan {}
621        fn only_built_by_new<T: OnlyBuiltByNew>() {}
622        only_built_by_new::<BlockPlan>();
623
624        // Otherwise a block could satisfy the framework with an empty plan
625        // and then render whatever it liked.
626        let err = BlockPlan::new(Vec::new()).unwrap_err().to_string();
627        assert!(err.contains(CONTRACT_BUG), "{err}");
628        assert!(err.contains("no outputs"), "{err}");
629    }
