10/// The list of paths `find_file` checks, in the order it checks them.
11///
12/// - An absolute path is tried as given.
13/// - A relative path is tried inside XDG_CONFIG_HOME (e.g. `~/.config`), then
14///   XDG_DATA_HOME (e.g. `~/.local/share/`), then `/usr/share/`, each with the
15///   `i3status-rust` directory (and `subdir`, if given) appended.
16/// - A path without an extension is also tried with `extension` appended.
17pub fn file_candidates(file: &str, subdir: Option<&str>, extension: Option<&str>) -> Vec<PathBuf> {
18    let file = Path::new(file);
19
20    let mut bases: Vec<PathBuf> = Vec::new();
21    if file.is_absolute() {
22        bases.push(file.into());
23    } else {
24        for dir in [config_dir(), data_dir(), Some("/usr/share".into())]
25            .into_iter()
26            .flatten()
27        {
28            let mut base: PathBuf = dir;
29            base.push("i3status-rust");
30            if let Some(subdir) = subdir {
31                base.push(subdir);
32            }
33            base.push(file);
34            bases.push(base);
35        }
36    }
37
38    let mut candidates = Vec::new();
39    for base in bases {
40        let with_extension = match (base.extension(), extension) {
41            (None, Some(extension)) => Some(base.with_extension(extension)),
42            _ => None,
43        };
44        candidates.push(base);
45        candidates.extend(with_extension);
46    }
47    candidates
48}

50/// Tries to find a file in standard locations (see [`file_candidates`]).
51pub fn find_file(
52    file: &str,
53    subdir: Option<&str>,
54    extension: Option<&str>,
55) -> Result<Option<PathBuf>> {
56    for candidate in file_candidates(file, subdir, extension) {
57        if candidate.try_exists().error("Unable to stat file")? {
58            return Ok(Some(candidate));
59        }
60    }
61    Ok(None)
62}

77where
78    T: DeserializeOwned,
79    P: AsRef<Path>,
80{
81    let path = path.as_ref();
82
83    let contents = std::fs::read_to_string(path)

103fn deserialize_toml_file_string<T>(contents: String, path: &Path) -> Result<T>
104where
105    T: DeserializeOwned,
106{
107    toml::from_str(&contents).map_err(|err| {
108        let location_msg = err
109            .span()
110            .map(|span| {
111                if span == (0..0) {
112                    String::new()
113                } else {
114                    let line = 1 + contents.as_bytes()[..(span.start)]
115                        .iter()
116                        .filter(|b| **b == b'\n')
117                        .count();
118                    format!(" at line {line}")
119                }
120            })
121            .unwrap_or_default();
122        Error::new(format!(
123            "Failed to deserialize TOML file {}{}: {}",
124            path.display(),
125            location_msg,
126            err.message()
127        ))
128    })
129}

131pub async fn read_file(path: impl AsRef<Path>) -> std::io::Result<String> {
132    let mut file = tokio::fs::File::open(path).await?;
133    let mut content = String::new();
134    file.read_to_string(&mut content).await?;
135    Ok(content.trim_end().to_string())
136}

150/// # Example
151///
152/// ```ignore
153/// let opt = Some(1);
154/// let m: HashMap<&'static str, String> = map! {
155///     "key" => "value",
156///     [if true] "hello" => "world",
157///     [if let Some(x) = opt] "opt" => x.to_string(),
158/// };
159/// map! { @extend m
160///     "new key" => "new value",
161///     "one" => "more",
162/// }
163/// ```
164#[macro_export]
165macro_rules! map {
166    (@extend $map:ident $( $([$($cond_tokens:tt)*])? $key:literal => $value:expr ),* $(,)?) => {{
167        $(
168        map!(@insert $map, $key, $value $(,$($cond_tokens)*)?);
169        )*
170    }};
171    (@extend $map:ident $( $key:expr => $value:expr ),* $(,)?) => {{
172        $(
173        map!(@insert $map, $key, $value);
174        )*
175    }};
176    (@insert $map:ident, $key:expr, $value:expr) => {{
177        $map.insert($key.into(), $value.into());
178    }};
179    (@insert $map:ident, $key:expr, $value:expr, if $cond:expr) => {{
180        if $cond {
181        $map.insert($key.into(), $value.into());
182        }
183    }};
184    (@insert $map:ident, $key:expr, $value:expr, if let $pat:pat = $match_on:expr) => {{
185        if let $pat = $match_on {
186        $map.insert($key.into(), $value.into());
187        }
188    }};
189    ($($tt:tt)*) => {{
190        #[allow(unused_mut)]
191        let mut m = ::std::collections::HashMap::new();
192        map!(@extend m $($tt)*);
193        m
194    }};
195}

220pub fn format_bar_graph(content: &[f64]) -> String {
221    // (x * one eighth block) https://en.wikipedia.org/wiki/Block_Elements
222    static BARS: [char; 8] = [
223        '\u{2581}', '\u{2582}', '\u{2583}', '\u{2584}', '\u{2585}', '\u{2586}', '\u{2587}',
224        '\u{2588}',
225    ];
226
227    // Find min and max
228    let mut min = f64::INFINITY;
229    let mut max = f64::NEG_INFINITY;
230    for &v in content {
231        min = min.min(v);
232        max = max.max(v);
233    }
234
235    let range = max - min;
236    content
237        .iter()
238        .map(|x| BARS[((x - min) / range * 7.).clamp(0., 7.) as usize])
239        .collect()
240}

242/// Doctor-only: when enabled (in doctor's worker process, never in the real
243/// bar), every country flag generated by [`country_flag_from_iso_code`] is
244/// recorded so the live block table can attribute the glyph to its block.
245static FLAG_RECORDER: std::sync::OnceLock<std::sync::Mutex<Vec<(String, String)>>> =
246    std::sync::OnceLock::new();
247
248pub(crate) fn enable_flag_recorder() {
249    let _ = FLAG_RECORDER.set(std::sync::Mutex::new(Vec::new()));
250}
251
252/// (country code, flag) pairs generated since the recorder was enabled.
253pub(crate) fn recorded_flags() -> Vec<(String, String)> {
254    let mut flags = FLAG_RECORDER
255        .get()
256        .and_then(|m| m.lock().ok().map(|v| v.clone()))
257        .unwrap_or_default();
258    flags.sort_unstable();
259    flags.dedup();
260    flags
261}
262
263fn record_flag(country_code: &str, flag: &str) {
264    if let Some(recorder) = FLAG_RECORDER.get()
265        && let Ok(mut recorder) = recorder.lock()
266    {
267        recorder.push((country_code.to_string(), flag.to_string()));
268    }
269}

271/// Convert 2 letter country code to Unicode
272pub fn country_flag_from_iso_code(country_code: &str) -> String {
273    let [mut b1, mut b2]: [u8; 2] = country_code.as_bytes().try_into().unwrap_or([0, 0]);
274
275    if !b1.is_ascii_uppercase() || !b2.is_ascii_uppercase() {
276        return country_code.into();
277    }
278
279    // Each char is encoded as 1F1E6 to 1F1FF for A-Z
280    b1 += 0xa5;
281    b2 += 0xa5;
282    // The last byte will always start with 101 (0xa0) and then the 5 least
283    // significant bits from the previous result
284    b1 = 0xa0 | (b1 & 0x1f);
285    b2 = 0xa0 | (b2 & 0x1f);
286    // Get the flag string from the UTF-8 representation of our Unicode characters.
287    let flag = String::from_utf8(vec![0xf0, 0x9f, 0x87, b1, 0xf0, 0x9f, 0x87, b2]).unwrap();
288    record_flag(country_code, &flag);
289    flag
290}

331pub trait StreamExtDebounced: futures::StreamExt {
332    fn next_debounced(&mut self) -> impl Future<Output = Option<Self::Item>>;
333}
334
335impl<T: futures::StreamExt + Unpin> StreamExtDebounced for T {
336    async fn next_debounced(&mut self) -> Option<Self::Item> {
337        let mut result = self.next().await?;
338        let mut noop_ctx = std::task::Context::from_waker(std::task::Waker::noop());
339        loop {
340            match self.poll_next_unpin(&mut noop_ctx) {
341                std::task::Poll::Ready(Some(x)) => result = x,
342                std::task::Poll::Ready(None) | std::task::Poll::Pending => return Some(result),
343            }
344        }
345    }
346}

348/// Run a future to completion synchronously.
349///
350/// If invoked outside an async runtime, a temporary current-thread runtime is used.
351/// If invoked within an active Tokio runtime, the future is driven in a dedicated
352/// thread to avoid blocking or nested-runtime issues.
353pub fn run_sync<F, T>(future: F) -> T
354where
355    F: std::future::Future<Output = T> + Send + 'static,
356    T: Send + 'static,
357{
358    match tokio::runtime::Handle::try_current() {
359        Ok(_) => std::thread::spawn(move || {
360            tokio::runtime::Builder::new_current_thread()
361                .enable_all()
362                .build()
363                .expect("failed to create tokio runtime")
364                .block_on(future)
365        })
366        .join()
367        .expect("worker thread panicked"),
368        Err(_) => tokio::runtime::Builder::new_current_thread()
369            .enable_all()
370            .build()
371            .expect("failed to create tokio runtime")
372            .block_on(future),
373    }
374}
